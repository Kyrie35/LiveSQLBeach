SELECT c.ContainRegistry, CAST(ABS(s.TempNowC - s.StoreTempC) AS REAL) / NULLIF(s.TempTolC, 0) * s.TempDevCount AS TBS, s.TempDevCount, GROUP_CONCAT(s.AlertKind) AS AlertTypes FROM Container c INNER JOIN SensorData s ON c.ContainRegistry = s.ContainLink WHERE CAST(ABS(s.TempNowC - s.StoreTempC) AS REAL) / NULLIF(s.TempTolC, 0) * s.TempDevCount > 2.0 AND s.TempDevCount > 5 GROUP BY c.ContainRegistry, s.TempNowC, s.StoreTempC, s.TempTolC, s.TempDevCount ORDER BY TBS DESC LIMIT 5;
