SELECT h.hubregistry, ( h.hubutilpct / 100.0 ) * ( h.storecapm3 / ( h.storeavailm3 + 1 ) ) AS resource_utilization_ratio, CASE WHEN ( h.hubutilpct / 100.0 ) * ( h.storecapm3 / ( h.storeavailm3 + 1 ) ) > 5 THEN 'High Utilization' WHEN ( h.hubutilpct / 100.0 ) * ( h.storecapm3 / ( h.storeavailm3 + 1 ) ) < 2 THEN 'Low Utilization' ELSE 'Moderate Utilization' END AS utilization_assessment FROM distributionhubs AS h ORDER BY resource_utilization_ratio DESC NULLS FIRST;
