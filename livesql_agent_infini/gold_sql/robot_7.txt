WITH cycle_metrics AS ( SELECT o.operbotdetref, SUM(o.progcyclecount) as tpc, AVG(o.cycletimesecval) as avg_cycle_time, SUM(o.progcyclecount)*1.0/NULLIF(SUM(o.cycletimesecval), 0) as oce, GROUP_CONCAT(DISTINCT o.currprogval) as unique_programs FROM operation o GROUP BY o.operbotdetref ), temporal_efficiency AS ( SELECT cm.operbotdetref, cm.oce, cm.tpc, cm.avg_cycle_time, LENGTH(cm.unique_programs) - LENGTH(REPLACE(cm.unique_programs, ',', '')) + 1 as program_count, FIRST_VALUE(o.cycletimesecval) OVER ( PARTITION BY o.operbotdetref ORDER BY rr.rects DESC ) as latest_cycle_time FROM cycle_metrics cm JOIN operation o ON cm.operbotdetref = o.operbotdetref JOIN robot_record rr ON o.operrecref = rr.recreg ) SELECT rd.botdetreg, rd.modelseriesval, te.oce, te.tpc, te.avg_cycle_time, te.program_count, te.latest_cycle_time, CASE WHEN te.oce < 100 AND te.tpc > 500000 THEN 'Low Efficiency' WHEN te.oce < 150 OR te.tpc > 300000 THEN 'Medium Efficiency' ELSE 'High Efficiency' END as efficiency_category, DENSE_RANK() OVER ( PARTITION BY rd.modelseriesval ORDER BY te.oce DESC ) as efficiency_rank_in_model FROM robot_details rd JOIN temporal_efficiency te ON rd.botdetreg = te.operbotdetref WHERE te.tpc > 100000 ORDER BY te.oce DESC, te.tpc DESC LIMIT 100;
