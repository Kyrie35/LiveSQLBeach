WITH spread_calc AS ( SELECT ( quote_depth_snapshot -> '$.metadata' ->> '$.mktcombo' ) AS market_pair, ( CAST(( quote_depth_snapshot -> '$.quotes' ->> '$.askquote' ) AS REAL) - CAST(( quote_depth_snapshot -> '$.quotes' ->> '$.bidquote' ) AS REAL) ) / CAST(( quote_depth_snapshot -> '$.quotes' ->> '$.midquote' ) AS REAL) * 100 AS spread_pct, marketdatanode FROM marketdata ), sentiment_data AS ( SELECT mdataref, ( market_sentiment_indicators -> '$.momentum' ->> '$.mktfeel' ) AS sentiment, CAST(( market_sentiment_indicators -> '$.momentum' ->> '$.buyforce' ) AS REAL) AS buy_force FROM analyticsindicators ) SELECT s.market_pair, s.spread_pct, d.sentiment, d.buy_force, AVG(s.spread_pct) OVER (PARTITION BY d.sentiment) AS avg_spread_by_sentiment, PERCENT_RANK() OVER (ORDER BY s.spread_pct NULLS LAST) AS spread_percentile FROM spread_calc AS s JOIN sentiment_data AS d ON s.marketdatanode = d.mdataref ORDER BY s.spread_pct DESC NULLS FIRST;
