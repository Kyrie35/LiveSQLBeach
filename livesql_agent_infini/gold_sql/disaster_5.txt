SELECT e.envhealthregistry, d.distregistry, ROUND( e.waterqualityindex * 0.4 + e.sanitationcoverage * 0.3 + e.vaccinationcoverage * 0.3, 2 ) AS public_health_resilience_score, CASE WHEN e.diseaserisk = 'High' AND e.waterqualityindex < 50 AND ( e.waterqualityindex * 0.4 + e.sanitationcoverage * 0.3 + e.vaccinationcoverage * 0.3 ) < 40 THEN 'Public Health Emergency' ELSE 'Standard Health Status' END AS health_status FROM environmentandhealth AS e JOIN disasterevents AS d ON e.envdistref = d.distregistry ORDER BY public_health_resilience_score NULLS LAST;
