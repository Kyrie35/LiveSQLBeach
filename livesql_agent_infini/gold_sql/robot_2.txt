WITH Degradation AS ( SELECT jc.jcdetref, SUM( (jc.j1tempval / MAX(jc.j1tempval, jc.j2tempval, jc.j3tempval, jc.j4tempval, jc.j5tempval, jc.j6tempval)) + (jc.j2tempval / MAX(jc.j1tempval, jc.j2tempval, jc.j3tempval, jc.j4tempval, jc.j5tempval, jc.j6tempval)) + (jc.j3tempval / MAX(jc.j1tempval, jc.j2tempval, jc.j3tempval, jc.j4tempval, jc.j5tempval, jc.j6tempval)) + (jc.j4tempval / MAX(jc.j1tempval, jc.j2tempval, jc.j3tempval, jc.j4tempval, jc.j5tempval, jc.j6tempval)) + (jc.j5tempval / MAX(jc.j1tempval, jc.j2tempval, jc.j3tempval, jc.j4tempval, jc.j5tempval, jc.j6tempval)) + (jc.j6tempval / MAX(jc.j1tempval, jc.j2tempval, jc.j3tempval, jc.j4tempval, jc.j5tempval, jc.j6tempval)) + jc.j1vibval + jc.j2vibval + jc.j3vibval + jc.j4vibval + jc.j5vibval + jc.j6vibval ) / (COUNT(*) * 6) AS jdi, MAX(jc.j1tempval, jc.j2tempval, jc.j3tempval, jc.j4tempval, jc.j5tempval, jc.j6tempval) AS mjt FROM joint_condition jc GROUP BY jc.jcdetref ), OperatingHours AS ( SELECT o.operbotdetref, MAX(o.totopshrval) AS toh FROM operation o GROUP BY o.operbotdetref ), RegressionData AS ( SELECT d.jcdetref, d.jdi, oh.toh FROM Degradation d INNER JOIN OperatingHours oh ON d.jcdetref = oh.operbotdetref ), AvgJDI AS (SELECT AVG(jdi) AS avg_jdi FROM RegressionData), AvgTOH AS (SELECT AVG(toh) AS avg_toh FROM RegressionData) SELECT d.jcdetref, d.jdi, d.mjt, oh.toh, SUM((rd.jdi - (SELECT avg_jdi FROM AvgJDI)) * (rd.toh - (SELECT avg_toh FROM AvgTOH))) / SUM((rd.toh - (SELECT avg_toh FROM AvgTOH)) * (rd.toh - (SELECT avg_toh FROM AvgTOH))) FROM Degradation d INNER JOIN OperatingHours oh ON d.jcdetref = oh.operbotdetref INNER JOIN RegressionData rd ON d.jcdetref = rd.jcdetref WHERE d.jdi > 1.5 AND d.mjt > 65 AND oh.toh > 5000 GROUP BY d.jcdetref, d.jdi, d.mjt, oh.toh ORDER BY d.jdi DESC LIMIT 100;
