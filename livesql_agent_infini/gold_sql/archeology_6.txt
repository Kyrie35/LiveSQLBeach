WITH mesh_analysis AS ( SELECT sm.zoneref AS site_code, sm.facetregistry AS mesh_id, ( sm.facetfaces / ( sm.facetverts * POWER(sm.facetresmm, 2) ) ) * 1000 AS mcr, sm.facetresmm, sm.geomdeltamm, CASE WHEN ( sm.facetfaces / ( sm.facetverts * POWER(sm.facetresmm, 2) ) ) * 1000 > 5.0 AND sm.facetresmm < 1.0 AND sm.geomdeltamm < 0.5 THEN TRUE ELSE FALSE END AS is_high_fidelity FROM scanmesh AS sm WHERE sm.facetverts > 0 AND sm.facetfaces > 0 ) SELECT s.zoneregistry AS site_code, s.zonelabel AS site_name, COUNT(ma.mesh_id) AS total_meshes, SUM(CASE WHEN ma.is_high_fidelity THEN 1 ELSE 0 END) AS high_fidelity_count, ROUND( ( CAST(SUM(CASE WHEN ma.is_high_fidelity THEN 1 ELSE 0 END) AS REAL) / NULLIF(COUNT(ma.mesh_id), 0) * 100 ), 2 ) AS high_fidelity_percentage, ROUND(CAST(AVG(ma.mcr) AS REAL), 2) AS avg_mcr, ROUND(CAST(AVG(ma.facetresmm) AS REAL), 2) AS avg_mesh_resolution_mm, ROUND(CAST(AVG(ma.geomdeltamm) AS REAL), 2) AS avg_geometric_accuracy_mm, CASE WHEN COUNT(ma.mesh_id) = 0 THEN 'No Mesh Data' WHEN SUM(CASE WHEN ma.is_high_fidelity THEN 1 ELSE 0 END) > 0 THEN 'Has High-Fidelity Meshes' ELSE 'Standard Mesh Quality' END AS mesh_quality_classification FROM sites AS s LEFT JOIN mesh_analysis AS ma ON s.zoneregistry = ma.site_code GROUP BY s.zoneregistry, s.zonelabel ORDER BY high_fidelity_percentage DESC NULLS FIRST, high_fidelity_count DESC NULLS FIRST;
