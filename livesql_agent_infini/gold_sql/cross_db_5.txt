WITH NonCompliantFlows AS ( SELECT com.ComplianceTrace, com.GdprComp, com.LocLawComp, df.OrigNation, df.DestNation FROM Compliance AS com RIGHT JOIN DataFlow AS df ON com.RecordRegistry = df.RecordRegistry WHERE EXISTS( SELECT 1 FROM DataFlow AS df2 WHERE df2.RecordRegistry = df.RecordRegistry AND df2.OrigNation <> df2.DestNation ) AND ( com.GdprComp = 'Non-compliant' OR com.LocLawComp = 'Non-compliant' ) ) SELECT ComplianceTrace, GdprComp, LocLawComp, OrigNation || ' -> ' || DestNation AS transfer_path FROM NonCompliantFlows ORDER BY ComplianceTrace NULLS LAST LIMIT 10;
