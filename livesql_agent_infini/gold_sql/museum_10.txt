WITH erf_calc AS ( SELECT a.ArtRegistry, a.ArtName, AVG(CASE WHEN LOWER(TRIM(s.EnvSensitivity))='high' THEN 10 WHEN LOWER(TRIM(s.EnvSensitivity))='medium' THEN 5 WHEN LOWER(TRIM(s.EnvSensitivity))='low' THEN 1 END) AS ERF FROM ArtifactsCore a LEFT JOIN SensitivityData s ON s.ArtRef = a.ArtRegistry GROUP BY a.ArtRegistry, a.ArtName ), high_sens AS ( SELECT s.ArtRef, json_group_array(json_object('type','EnvSensitivity','value', s.EnvSensitivity)) AS high_sensitivities FROM SensitivityData s WHERE LOWER(TRIM(s.EnvSensitivity))='high' GROUP BY s.ArtRef ) SELECT ec.ArtRegistry, ec.ArtName, ROUND(ec.ERF,2) AS ERF, COALESCE(hs.high_sensitivities,'[]') AS high_sensitivities FROM erf_calc ec LEFT JOIN high_sens hs ON hs.ArtRef = ec.ArtRegistry WHERE ec.ERF > 7 ORDER BY ec.ERF DESC;
