WITH ds_cte AS ( SELECT physregistry, ( ( drophtm / 2.0 ) + ( bendforce / 100.0 ) + ( twistdeg / 90.0 ) ) * ( usbconndur / 10000.0 ) * 10 AS ds, dustres, waterres, bendforce FROM physicaldurability ) SELECT ds.physregistry, ROUND(ds.ds, 2) AS ds, CASE WHEN COALESCE(ds.dustres, '') = 'IPX0' THEN 0 WHEN COALESCE(ds.dustres, '') = 'IPX1' THEN 1 WHEN COALESCE(ds.dustres, '') = 'IPX2' THEN 2 WHEN COALESCE(ds.dustres, '') = 'IPX3' THEN 3 ELSE 0 END AS dustres_score, CASE WHEN COALESCE(ds.waterres, '') = 'IPX0' THEN 0 WHEN COALESCE(ds.waterres, '') = 'IPX1' THEN 1 WHEN COALESCE(ds.waterres, '') = 'IPX2' THEN 2 WHEN COALESCE(ds.waterres, '') = 'IPX3' THEN 3 ELSE 0 END AS waterres_score, ds.bendforce, ROUND( ds.ds * ( 1 + ( CASE WHEN COALESCE(ds.dustres, '') = 'IPX0' THEN 0 WHEN COALESCE(ds.dustres, '') = 'IPX1' THEN 1 WHEN COALESCE(ds.dustres, '') = 'IPX2' THEN 2 WHEN COALESCE(ds.dustres, '') = 'IPX3' THEN 3 ELSE 0 END + CASE WHEN COALESCE(ds.waterres, '') = 'IPX0' THEN 0 WHEN COALESCE(ds.waterres, '') = 'IPX1' THEN 1 WHEN COALESCE(ds.waterres, '') = 'IPX2' THEN 2 WHEN COALESCE(ds.waterres, '') = 'IPX3' THEN 3 ELSE 0 END ) / 6.0 ) * ( 1 - ( 100 - ds.bendforce ) / 200.0 ), 2 ) AS per FROM ds_cte AS ds WHERE ds.ds * ( 1 + ( CASE WHEN COALESCE(ds.dustres, '') = 'IPX0' THEN 0 WHEN COALESCE(ds.dustres, '') = 'IPX1' THEN 1 WHEN COALESCE(ds.dustres, '') = 'IPX2' THEN 2 WHEN COALESCE(ds.dustres, '') = 'IPX3' THEN 3 ELSE 0 END + CASE WHEN COALESCE(ds.waterres, '') = 'IPX0' THEN 0 WHEN COALESCE(ds.waterres, '') = 'IPX1' THEN 1 WHEN COALESCE(ds.waterres, '') = 'IPX2' THEN 2 WHEN COALESCE(ds.waterres, '') = 'IPX3' THEN 3 ELSE 0 END ) / 6.0 ) * ( 1 - ( 100 - ds.bendforce ) / 200.0 ) > 9.0 ORDER BY per DESC NULLS FIRST;
