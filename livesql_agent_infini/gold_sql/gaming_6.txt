WITH wpr_cte AS ( SELECT d.devregistry, ( d.wlrangem / 10.0 ) * ( 1 - d.wllatvar / 5.0 ) * ( 1 + ( CASE WHEN d.wlchanhop THEN 1 ELSE 0 END ) / 2.0 ) * ( ( t.wlsignal + 100 ) / 100.0 ) AS wpr, d.wllatvar FROM deviceidentity AS d JOIN testsessions AS t ON d.devsessionref = t.sessionregistry ), ber_cte AS ( SELECT d.devregistry, ( t.battlifeh * t.battcapmah ) / ( d.pwractmw * 10.0 ) AS ber FROM deviceidentity AS d JOIN testsessions AS t ON d.devsessionref = t.sessionregistry WHERE d.pwractmw > 0 ) SELECT wpr.devregistry, ROUND(wpr.wpr * SQRT(ber.ber / 5.0) * ( 1 - wpr.wllatvar / 3.0 ) * 2, 2) AS wpe FROM wpr_cte AS wpr JOIN ber_cte AS ber ON wpr.devregistry = ber.devregistry WHERE wpr.wpr * SQRT(ber.ber / 5.0) * ( 1 - wpr.wllatvar / 3.0 ) * 2 > 8.5 ORDER BY wpe DESC NULLS FIRST;
