WITH VendorVTI AS ( SELECT vendregistry, mktref, vendtxcount, CASE WHEN vendtxcount > 0 THEN ( CAST(vendsucccount AS REAL) / vendtxcount * 100 ) - ( CAST(venddisputecount AS REAL) / vendtxcount * 50 ) + ( vendrate * 5 ) ELSE 0 END AS vti FROM vendors ), VendorMarketInfo AS ( SELECT v.vendregistry, v.vti, v.vendtxcount, m.sizecluster, 1 AS distinct_market_count FROM VendorVTI AS v LEFT JOIN markets AS m ON v.mktref = m.mktregistry ) SELECT vmi.vendregistry, vmi.sizecluster, vmi.vti, ( vmi.distinct_market_count * 5 ) + ( CAST(vmi.vendtxcount AS REAL) / 50 ) + ( vmi.vti * 0.1 ) - ( 1 - CASE vmi.sizecluster WHEN 'Small' THEN 1 WHEN 'Medium' THEN 2 WHEN 'Large' THEN 3 WHEN 'Mega' THEN 4 ELSE 0 END ) * 10 AS vnc FROM VendorMarketInfo AS vmi WHERE ( vmi.distinct_market_count * 5 ) + ( CAST(vmi.vendtxcount AS REAL) / 50 ) + ( vmi.vti * 0.1 ) - ( 1 - CASE vmi.sizecluster WHEN 'Small' THEN 1 WHEN 'Medium' THEN 2 WHEN 'Large' THEN 3 WHEN 'Mega' THEN 4 ELSE 0 END ) * 10 > 50 ORDER BY vnc DESC NULLS FIRST;
