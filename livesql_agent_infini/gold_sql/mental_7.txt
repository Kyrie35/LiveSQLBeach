WITH FacilityAdherence AS ( SELECT e.facid, CASE WHEN COUNT(tro.txoutkey) > 0 THEN CAST(SUM(CASE WHEN tro.txadh IN ('High', 'Medium') THEN 1 ELSE 0 END) AS REAL) / COUNT(tro.txoutkey) ELSE 0 END AS tar FROM treatmentoutcomes tro JOIN treatmentbasics tb ON tro.txref = tb.txkey JOIN encounters e ON tb.encref = e.enckey WHERE e.facid IS NOT NULL GROUP BY e.facid ), FacilityPatientCount AS ( SELECT e.facid, COUNT(DISTINCT e.patref) AS total_patients FROM encounters e WHERE e.facid IS NOT NULL GROUP BY e.facid ), FacilityStabilityMetrics AS ( SELECT e.facid, COALESCE(SUM(COALESCE(tb.crisisint, 0)) / NULLIF(fpc.total_patients, 0), 0) AS cif, COALESCE(SUM(COALESCE(e.missappt, 0)) / NULLIF(fpc.total_patients, 0), 0) AS mar FROM encounters e LEFT JOIN treatmentbasics tb ON e.enckey = tb.encref JOIN FacilityPatientCount fpc ON e.facid = fpc.facid WHERE e.facid IS NOT NULL GROUP BY e.facid, fpc.total_patients ), FacilityPSM AS ( SELECT facid, cif, mar, 1.0 / (1.0 + cif + mar) AS psm FROM FacilityStabilityMetrics ), FacilityMetrics AS ( SELECT fa.facid, fa.tar, fp.psm FROM FacilityAdherence fa JOIN FacilityPSM fp ON fa.facid = fp.facid ), Thresholds AS ( SELECT ( SELECT tar FROM ( SELECT tar FROM FacilityMetrics ORDER BY tar LIMIT 1 OFFSET (SELECT (COUNT(*) - 1) / 2 FROM FacilityMetrics) ) ) AS median_tar, ( SELECT psm FROM ( SELECT psm FROM FacilityMetrics ORDER BY psm LIMIT 1 OFFSET (SELECT (COUNT(*) - 1) / 2 FROM FacilityMetrics) ) ) AS median_psm ) SELECT fm.facid, ROUND(fm.tar, 2) AS tar, ROUND(fm.psm, 2) AS psm, CASE WHEN fm.tar >= t.median_tar AND fm.psm >= t.median_psm THEN 'High Adherence, High Stability' WHEN fm.tar >= t.median_tar AND fm.psm < t.median_psm THEN 'High Adherence, Low Stability' WHEN fm.tar < t.median_tar AND fm.psm >= t.median_psm THEN 'Low Adherence, High Stability' ELSE 'Low Adherence, Low Stability' END AS performance_quadrant FROM FacilityMetrics fm, Thresholds t ORDER BY performance_quadrant, fm.facid LIMIT 100;
