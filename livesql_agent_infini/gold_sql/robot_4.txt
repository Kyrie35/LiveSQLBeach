WITH controller_metrics AS ( SELECT systemoverseerrobot, AVG( CAST(( controller_metrics ->> '$.load_value' ) AS REAL) + CAST(( controller_metrics ->> '$.thermal_level' ) AS REAL) ) AS csi FROM system_controller GROUP BY systemoverseerrobot ), operation_count AS ( SELECT operbotdetref, COUNT(*) AS operation_count FROM operation GROUP BY operbotdetref ) SELECT rd.botdetreg, cm.csi, oc.operation_count, CASE WHEN cm.csi > 100 AND oc.operation_count > 2 THEN 'HIGH RISK' ELSE 'NORMAL' END AS controller_status FROM robot_details AS rd JOIN controller_metrics AS cm ON rd.botdetreg = cm.systemoverseerrobot JOIN operation_count AS oc ON rd.botdetreg = oc.operbotdetref ORDER BY cm.csi DESC NULLS FIRST LIMIT 100;
