SELECT d.devregistry, d.makername, d.modnum, t.pollratehz, t.inplagms, t.resptimems, ROUND( ( t.pollratehz / 100.0 ) * ( 1 - ( t.inplagms + t.resptimems ) / 30.0 ) * 10, 2 ) AS irs FROM deviceidentity AS d JOIN testsessions AS t ON d.devsessionref = t.sessionregistry WHERE ( t.pollratehz / 100.0 ) * ( 1 - ( t.inplagms + t.resptimems ) / 30.0 ) * 10 > 8.0 AND t.devscope IN ('Keyboard', 'Mouse') ORDER BY irs DESC NULLS FIRST;
