WITH environment_analysis AS ( SELECT se.zoneref AS site_code, se.ambictemp, se.humepct, se.illumelux, 100 - 2.5 * ABS(se.ambictemp - 20) - POWER(ABS(( se.humepct - 50 ) / 2), 1.5) - 600 / ( se.illumelux + 100 ) AS esi FROM scanenvironment AS se ) SELECT s.zoneregistry AS site_code, s.zonelabel AS site_name, ROUND(CAST(AVG(ea.ambictemp) AS REAL), 1) AS avg_temperature_c, ROUND(CAST(AVG(ea.humepct) AS REAL), 1) AS avg_humidity_pct, ROUND(CAST(AVG(ea.illumelux) AS REAL), 1) AS avg_illumination_lux, ROUND(CAST(AVG(ea.esi) AS REAL), 1) AS avg_esi, NTILE(4) OVER (ORDER BY AVG(ea.esi) NULLS LAST) AS esi_quartile, CASE WHEN AVG(ea.esi) > 85 THEN 'Optimal Scanning Conditions' WHEN AVG(ea.esi) > 70 THEN 'Good Scanning Conditions' WHEN AVG(ea.esi) > 50 THEN 'Acceptable Scanning Conditions' ELSE 'Challenging Scanning Conditions' END AS scanning_condition_class FROM sites AS s JOIN environment_analysis AS ea ON s.zoneregistry = ea.site_code GROUP BY s.zoneregistry, s.zonelabel ORDER BY avg_esi DESC NULLS FIRST;
