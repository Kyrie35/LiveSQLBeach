SELECT f.nicklabel, f.tierstep, ROUND( CAST(( e.engrate * 0.4 + e.socintscore / 100 * 0.3 + e.actdayswk / 7.0 * 0.2 + e.avgsesscount / 10.0 * 0.1 ) AS REAL), 2 ) AS fei, ROUND(CAST(( ms.spendusd * ( 1 + ms.membdays / 365.0 * 0.5 ) ) AS REAL), 2) AS mv, CASE WHEN f.tierstep >= 8 AND ( e.engrate * 0.4 + e.socintscore / 100 * 0.3 + e.actdayswk / 7.0 * 0.2 + e.avgsesscount / 10.0 * 0.1 ) > 0.7 AND ms.spendusd * ( 1 + ms.membdays / 365.0 * 0.5 ) > 200 THEN 'Superfan' ELSE 'Regular Fan' END AS fan_classification FROM fans AS f JOIN membershipandspending AS ms ON ms.memberfanpivot = f.userregistry JOIN engagement AS e ON e.engagememberpivot = ms.memberreg WHERE f.statustag = 'Active' OR f.statustag = 'VIP' ORDER BY fei DESC NULLS FIRST, mv DESC NULLS FIRST;
