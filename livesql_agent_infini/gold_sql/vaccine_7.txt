WITH CoolantStatus AS (SELECT c.ContainRegistry, c.CoolRemainPct, (100.0 - c.CoolRemainPct) / NULLIF(julianday('now') - julianday(c.RefillLatest), 0) AS CDR, DENSE_RANK() OVER (ORDER BY (100.0 - c.CoolRemainPct) / NULLIF(julianday('now') - julianday(c.RefillLatest), 0) DESC) AS DepletionRank FROM Container c WHERE c.RefillLatest IS NOT NULL) SELECT ContainRegistry, CoolRemainPct, ROUND(CDR, 2) AS CDR, DepletionRank FROM CoolantStatus WHERE CDR IS NOT NULL ORDER BY DepletionRank LIMIT 5;
