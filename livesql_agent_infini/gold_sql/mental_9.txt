WITH FacilitySRP AS ( SELECT e.facid, COUNT(*) AS total_assessments, COUNT(*) FILTER(WHERE asr.suicrisk IN ('High', 'Severe')) AS high_risk_assessments, CASE WHEN COUNT(*) > 0 THEN ( CAST(COUNT(*) FILTER(WHERE asr.suicrisk IN ('High', 'Severe')) AS REAL) * 100 / COUNT(*) ) ELSE 0 END AS srp FROM assessmentsymptomsandrisk AS asr JOIN assessmentbasics AS ab ON asr.asrkey = ab.abkey JOIN encounters AS e ON ab.abkey = e.abref WHERE NOT e.facid IS NULL GROUP BY e.facid ), HighSRPFacilities AS ( SELECT facid FROM FacilitySRP WHERE srp > 20.0 ), FacilityPFIS AS ( SELECT e.facid, ROUND( AVG( CASE asd.funcimp WHEN 'Severe' THEN 3 WHEN 'Moderate' THEN 2 WHEN 'Mild' THEN 1 ELSE 0 END ) ) AS pfis FROM assessmentsocialanddiagnosis AS asd JOIN assessmentbasics AS ab ON asd.asdkey = ab.abkey JOIN encounters AS e ON ab.abkey = e.abref WHERE e.facid IN ( SELECT facid FROM HighSRPFacilities ) AND NOT asd.funcimp IS NULL GROUP BY e.facid ), FacilityFRAI AS ( SELECT f.fackey AS facid, ROUND( AVG( CASE ( f.support_and_resources ->> '$.community_resources' ) WHEN 'Comprehensive' THEN 3 WHEN 'Adequate' THEN 2 WHEN 'Limited' THEN 1 ELSE 0 END ) ) AS frai FROM facilities AS f WHERE f.fackey IN ( SELECT facid FROM HighSRPFacilities ) AND NOT f.support_and_resources ->> '$.community_resources' IS NULL GROUP BY f.fackey ) SELECT hspf.facid, fpf.pfis, ffr.frai, ( fpf.pfis - ffr.frai ) AS rdd FROM HighSRPFacilities AS hspf JOIN FacilityPFIS AS fpf ON hspf.facid = fpf.facid JOIN FacilityFRAI AS ffr ON hspf.facid = ffr.facid ORDER BY rdd DESC NULLS FIRST LIMIT 100;
