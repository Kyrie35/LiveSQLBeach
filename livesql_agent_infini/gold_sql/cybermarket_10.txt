SELECT vendregistry, ROUND( CASE vendchecklvl WHEN 'Basic' THEN 1 WHEN 'Advanced' THEN 2 WHEN 'Premium' THEN 3 ELSE 0 END * 15 + ( vendrate * 10 ) + ( CAST(vendsucccount AS REAL) / vendtxcount * 50 ) - ( CAST(venddisputecount AS REAL) / vendtxcount * 100 ), 2 ) AS vrs FROM vendors WHERE vendtxcount > 0 ORDER BY vrs DESC NULLS FIRST;
