WITH PeerMimicryTraders AS ( SELECT DISTINCT tr.trdref AS tradereg FROM advancedbehavior AS ab JOIN transactionrecord AS tr ON ab.translink = tr.transreg WHERE ABS(ab.patsim - ab.peercorr) < 0.1 AND ab.peercorr > 0.7 ), CollusionNetworkCases AS ( SELECT cc.compreg, cc.transref FROM investigationdetails AS idet JOIN compliancecase AS cc ON idet.compref = cc.compreg WHERE idet.tcirclesz > 5 AND idet.grpbehsc > 0.6 AND idet.commpat = 'Regular' ), NetworkMimicryRiskCases AS ( SELECT DISTINCT cnc.compreg FROM CollusionNetworkCases AS cnc JOIN transactionrecord AS tr ON cnc.transref = tr.transreg JOIN PeerMimicryTraders AS pmt ON tr.trdref = pmt.tradereg ), InvestigationScores AS ( SELECT nmrc.compreg, ROUND(( 0.6 * idet.behansc ) + ( 0.4 * idet.netansc ), 2) AS iii FROM NetworkMimicryRiskCases AS nmrc JOIN investigationdetails AS idet ON nmrc.compreg = idet.compref ) SELECT compreg, iii FROM InvestigationScores ORDER BY iii DESC NULLS FIRST LIMIT 100;
