WITH TherapyDetails AS ( SELECT e.patref AS patkey, tb.txkey, e.timemark AS encounter_time, tb.therapy_details ->> '$.type' AS therapy_type, tb.therapy_details ->> '$.changes' AS therapy_change, CASE ( tb.therapy_details ->> '$.engagement' ) WHEN 'High' THEN 3 WHEN 'Medium' THEN 2 WHEN 'Low' THEN 1 WHEN 'Non-compliant' THEN 0 ELSE NULL END AS engagement_score FROM treatmentbasics AS tb JOIN encounters AS e ON tb.encref = e.enckey WHERE NOT tb.therapy_details IS NULL ), EngagementChanges AS ( SELECT patkey, txkey, encounter_time, therapy_change, engagement_score, LAG(engagement_score, 1) OVER (PARTITION BY patkey ORDER BY encounter_time NULLS LAST) AS prev_engagement_score FROM TherapyDetails ) SELECT therapy_change, COUNT(*) AS total_records, ROUND(CAST(AVG(prev_engagement_score) AS REAL), 2) AS avg_previous_score, ROUND(CAST(AVG(engagement_score) AS REAL), 2) AS avg_current_score, ROUND( CAST(AVG( CASE WHEN NOT prev_engagement_score IS NULL THEN engagement_score - prev_engagement_score END ) AS REAL), 2 ) AS avg_engagement_score_change FROM EngagementChanges WHERE NOT therapy_change IS NULL GROUP BY therapy_change ORDER BY total_records DESC NULLS FIRST;
