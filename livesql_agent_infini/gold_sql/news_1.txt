WITH session_engagement AS ( SELECT seshkey, DATE(seshstart) AS sess_date, seshviews, engscore, seshdur, ( seshviews * CAST(engscore AS REAL) ) / seshdur AS user_engagement_rate FROM sessions ) SELECT sess_date, user_engagement_rate, AVG(user_engagement_rate) OVER (PARTITION BY sess_date) AS avg_daily_uer FROM session_engagement WHERE user_engagement_rate > ( SELECT CAST(AVG(( seshviews * CAST(engscore AS REAL) ) / seshdur) AS REAL) FROM sessions ) ORDER BY sess_date NULLS LAST;
