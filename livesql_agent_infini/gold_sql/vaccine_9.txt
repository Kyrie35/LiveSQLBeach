WITH MaintenanceStatus AS ( SELECT rm.RegMaintRegistry, rm.CompScore, rm.Incidents, rm.MaintDateNext, rm.CalibDateNext, COALESCE(rm.CompScore * (1 - COALESCE(rm.Incidents, 0) / 10), 0) AS MCS FROM RegulatoryAndMaintenance rm ) SELECT RegMaintRegistry, ROUND(MCS, 2) AS mcs, MAX( ( JULIANDAY(date('now')) - JULIANDAY(MaintDateNext) ), ( JULIANDAY(date('now')) - JULIANDAY(CalibDateNext) ), 0 ) AS days_overdue, 'Overdue' AS maintenance_status FROM MaintenanceStatus WHERE MCS < 0.7 AND (date('now') > MaintDateNext OR date('now') > CalibDateNext) AND MCS IS NOT NULL ORDER BY days_overdue DESC LIMIT 5;
