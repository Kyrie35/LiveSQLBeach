WITH fan_memberships AS ( SELECT f.userregistry, f.nicklabel, ms.memberreg, ms.spendusd, ms.membdays, ms.membkind FROM fans AS f JOIN membershipandspending AS ms ON ms.memberfanpivot = f.userregistry ), fan_interactions AS ( SELECT i.interactfanpivot AS userregistry, i.gifttot, CAST(( i.engagement_metrics -> '$.chat_activity' ->> '$.chatmsg' ) AS INTEGER) AS chatmsg, CAST(( i.engagement_metrics -> '$.session' ->> '$.watchhrs' ) AS REAL) AS watch_hours FROM interactions AS i ), silent_supporters AS ( SELECT fm.userregistry, fm.nicklabel, fm.spendusd * ( 1 + fm.membdays / 365.0 * 0.5 ) * ( 1 + COALESCE(fi.gifttot, 0) / 10.0 * 0.2 ) AS mv, e.engrate, fi.chatmsg, p.sesscount, CAST(COALESCE(fi.chatmsg, 0) AS REAL) / NULLIF(p.sesscount, 0) AS chat_message_ratio, e.contpref, fm.membkind, fm.membdays, fm.spendusd, fi.watch_hours FROM fan_memberships AS fm JOIN engagement AS e ON e.engagememberpivot = fm.memberreg LEFT JOIN fan_interactions AS fi ON fi.userregistry = fm.userregistry JOIN preferencesandsettings AS p ON p.preferencesmemberpivot = fm.memberreg WHERE fm.spendusd * ( 1 + fm.membdays / 365.0 * 0.5 ) * ( 1 + COALESCE(fi.gifttot, 0) / 10.0 * 0.2 ) > 100 AND e.engrate < 0.3 AND CAST(COALESCE(fi.chatmsg, 0) AS REAL) / NULLIF(p.sesscount, 0) < 0.5 ) SELECT nicklabel, mv, engrate, chat_message_ratio, membkind, spendusd / NULLIF(watch_hours, 0) AS revenue_per_hour, CASE WHEN contpref = 'Music' THEN 'Content Consumer (Music)' WHEN contpref = 'Dance' THEN 'Content Consumer (Dance)' WHEN contpref = 'Gaming' THEN 'Content Consumer (Gaming)' ELSE 'General Content Consumer' END AS viewer_category FROM silent_supporters ORDER BY mv DESC NULLS FIRST;
