WITH communication_security_risk AS ( SELECT commregistry, vpnflag, langpattern, msgtally, connpatscore, ( iptally * 5 ) + ( tornodecount * 2 ) + ( CASE vpnflag WHEN 'Yes' THEN 1 WHEN 'Suspected' THEN 0.5 ELSE 0 END * 30 ) + ( brwsrunique / 10 ) + ( susppatscore * 3 ) + ( riskindiccount * 4 ) AS csr FROM communication ) SELECT c.commregistry, c.vpnflag, c.langpattern, c.msgtally, ROUND(csr.csr, 2) AS communication_security_risk, ROUND( ( CASE c.langpattern WHEN 'Consistent' THEN 1 WHEN 'Variable' THEN 2 WHEN 'Suspicious' THEN 3 ELSE 0 END * 15 ) + ( csr.csr * 0.2 ) + ( c.msgtally * 0.5 ) - ( CASE WHEN c.vpnflag = 'Yes' THEN 0 WHEN c.vpnflag = 'Suspected' THEN 0.5 ELSE 1 END * 20 ), 2 ) AS communication_pattern_risk, CASE WHEN ( ( CASE c.langpattern WHEN 'Consistent' THEN 1 WHEN 'Variable' THEN 2 WHEN 'Suspicious' THEN 3 ELSE 0 END * 15 ) + ( csr.csr * 0.2 ) + ( c.msgtally * 0.5 ) - ( CASE WHEN c.vpnflag = 'Yes' THEN 0 WHEN c.vpnflag = 'Suspected' THEN 0.5 ELSE 1 END * 20 ) ) > 70 AND c.langpattern = 'Suspicious' AND csr.csr > 80 THEN 'Deceptive Pattern' ELSE 'Normal Pattern' END AS pattern_classification FROM communication AS c JOIN communication_security_risk AS csr ON c.commregistry = csr.commregistry ORDER BY communication_pattern_risk DESC NULLS FIRST;
