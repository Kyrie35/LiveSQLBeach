WITH RelevantTraders AS ( SELECT t.tradereg, t.voldaily / NULLIF(t.acctbal, 0) AS dtr, CAST(json_extract(t.trading_performance, '$.risklevel.levratio') AS REAL) AS tle, json_extract(t.trading_performance, '$.risklevel.risklevel') AS risk_level_text FROM trader t ), HighRiskProfile AS ( SELECT tradereg FROM RelevantTraders WHERE (tle > 5.0 AND risk_level_text = 'Aggressive') OR dtr > 0.5 ), RecentSignificantEnforcements AS ( SELECT DISTINCT tr.trdref AS tradereg FROM enforcementactions ea JOIN compliancecase cc ON ea.compref2 = cc.compreg JOIN transactionrecord tr ON cc.transref = tr.transreg WHERE (ea.penimp IN ('Fine', 'Ban') OR ea.acttake = 'Suspension' OR ea.busrestr = 'Full') AND cc.lastrevdt >= date((SELECT MAX(lastrevdt) FROM compliancecase), '-12 months') ) SELECT DISTINCT hrp.tradereg FROM HighRiskProfile hrp INNER JOIN RecentSignificantEnforcements rse ON hrp.tradereg = rse.tradereg LIMIT 100;
