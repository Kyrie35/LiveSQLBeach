WITH TraderPerformance AS ( SELECT tradereg, CAST(json_extract(trading_performance, '$.winpct') AS REAL) AS winpct, MAX(1, CAST(json_extract(trading_performance, '$.risklevel.levratio') AS REAL)) AS tle_floor1 FROM trader ), RAWR_Calculation AS ( SELECT tradereg, winpct / tle_floor1 AS rawr FROM TraderPerformance ), PeerMimicryFlags AS ( SELECT DISTINCT tr.trdref as tradereg, CASE WHEN ABS(ab.patsim - ab.peercorr) < 0.1 AND ab.peercorr > 0.7 THEN TRUE ELSE FALSE END as is_mimicry_suspect FROM advancedbehavior ab JOIN transactionrecord tr ON ab.translink = tr.transreg ) SELECT COALESCE(pmf.is_mimicry_suspect, FALSE) AS is_mimicry_suspect_group, ROUND(AVG(rc.rawr), 2) AS average_rawr FROM RAWR_Calculation rc LEFT JOIN PeerMimicryFlags pmf ON rc.tradereg = pmf.tradereg GROUP BY COALESCE(pmf.is_mimicry_suspect, FALSE) LIMIT 100;
