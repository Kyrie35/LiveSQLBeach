WITH fan_memberships AS ( SELECT f.userregistry, f.nicklabel, ms.memberreg, ms.spendusd, ms.membdays, COALESCE(i.gifttot, 0) AS gifttot FROM fans f JOIN membershipandspending ms ON ms.memberfanpivot = f.userregistry LEFT JOIN interactions i ON i.interactfanpivot = f.userregistry ), fan_engagement AS ( SELECT fm.userregistry, fm.nicklabel, e.engrate * 0.4 + e.socintscore / 100 * 0.3 + e.actdayswk / 7.0 * 0.2 + e.avgsesscount / 10.0 * 0.1 AS fei, fm.spendusd * (1 + fm.membdays / 365.0 * 0.5) * (1 + fm.gifttot / 10.0 * 0.2) AS mv, e.engagereg FROM fan_memberships fm JOIN engagement e ON e.engagememberpivot = fm.memberreg ), retention_risk AS ( SELECT fe.userregistry, fe.nicklabel, fe.fei, fe.mv, CASE WHEN r.churnflag = 'None' THEN 0 WHEN r.churnflag = 'Low' THEN 1 WHEN r.churnflag = 'Medium' THEN 2 WHEN r.churnflag = 'High' THEN 3 END AS rrf_approx FROM fan_engagement fe JOIN retentionandinfluence r ON r.retainengagepivot = fe.engagereg ), fan_lifetime_value AS ( SELECT userregistry, nicklabel, mv * (1 - rrf_approx / 10.0) * (1 + fei) * 24 AS flv FROM retention_risk ), percentile_values AS ( SELECT (SELECT flv FROM fan_lifetime_value ORDER BY flv LIMIT 1 OFFSET (SELECT COUNT(*) * CAST((SELECT COUNT(*) FROM fan_lifetime_value) * 0.5 AS INT))) AS median, (SELECT flv FROM fan_lifetime_value ORDER BY flv LIMIT 1 OFFSET (SELECT COUNT(*) * CAST((SELECT COUNT(*) FROM fan_lifetime_value) * 0.75 AS INT))) AS p75, (SELECT flv FROM fan_lifetime_value ORDER BY flv LIMIT 1 OFFSET (SELECT COUNT(*) * CAST((SELECT COUNT(*) FROM fan_lifetime_value) * 0.9 AS INT))) AS p90 FROM fan_lifetime_value ) SELECT f.userregistry, f.nicklabel, ROUND(f.flv, 2) AS flv, CASE WHEN f.flv > p.p90 THEN 'Top Tier' WHEN f.flv > p.p75 THEN 'High Value' WHEN f.flv > p.median THEN 'Medium Value' ELSE 'Low Value' END AS value_segment FROM fan_lifetime_value f, percentile_values p ORDER BY f.flv DESC;
