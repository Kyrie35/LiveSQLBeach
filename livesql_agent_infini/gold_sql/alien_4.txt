WITH tech_signals AS ( SELECT s.SignalRegistry, s.TelescRef, p.TechSigProb * ( 1 - p.NatSrcProb ) * p.SigUnique * ( 0.5 + p.AnomScore / 10 ) AS TOLS, s.BwHz / ( s.CenterFreqMhz * 1000000 ) AS BFR FROM Signals AS s JOIN SignalProbabilities AS p ON s.SignalRegistry = p.SignalRef WHERE p.TechSigProb > 0.7 AND p.NatSrcProb < 0.3 AND p.ArtSrcProb < 50 AND s.BwHz / ( s.CenterFreqMhz * 1000000 ) < 0.001 ) SELECT t.ObservStation, COUNT(*) AS potential_tech_signals, AVG(ts.TOLS) AS avg_tols, AVG(ts.BFR) AS avg_bfr, COUNT(*) * 100.0 / ( SELECT COUNT(*) FROM tech_signals ) AS percentage_of_total FROM tech_signals AS ts JOIN Telescopes AS t ON ts.TelescRef = t.TelescRegistry GROUP BY t.ObservStation ORDER BY potential_tech_signals DESC NULLS FIRST;
