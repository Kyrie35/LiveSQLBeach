SELECT t.transportregistry, ROUND( ( t.totaldeliverytons / h.hubcaptons ) * t.deliverysuccessrate * ( 1 - t.vehiclebreakrate / 100.0 ) * 100, 2 ) AS logistics_performance_metric, CASE WHEN ( t.totaldeliverytons / h.hubcaptons ) * t.deliverysuccessrate * ( 1 - t.vehiclebreakrate / 100.0 ) * 100 < 30 AND t.lastmilestatus = 'Suspended' AND t.vehiclebreakrate > 15 THEN 'Logistics Breakdown' ELSE 'Normal Operation' END AS logistics_status FROM transportation AS t JOIN disasterevents AS d ON t.transportdistref = d.distregistry JOIN distributionhubs AS h ON t.transporthubref = h.hubregistry ORDER BY logistics_performance_metric NULLS LAST;
