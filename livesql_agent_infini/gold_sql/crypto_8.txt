WITH risk AS ( SELECT r.riskandmarginpivot, CAST(( r.risk_margin_profile -> '$.position' ->> '$.possum' ) AS REAL) AS possum, CAST(( r.risk_margin_profile -> '$.position' ->> '$.posmagn' ) AS REAL) AS posmagn, o.userlink FROM riskandmargin AS r JOIN orders AS o ON r.ordervault = o.recordvault ) SELECT riskandmarginpivot, possum, posmagn, CAST(ab.walletsum AS REAL), posmagn * ( possum / CAST(ab.walletsum AS REAL) ) AS effective_leverage FROM risk JOIN accountbalances AS ab ON risk.userlink = ab.usertag;
