WITH snapshot AS ( SELECT quote_depth_snapshot, CAST(( quote_depth_snapshot -> '$.quotes' ->> '$.askquote' ) AS REAL) AS askquote, CAST(( quote_depth_snapshot -> '$.quotes' ->> '$.bidquote' ) AS REAL) AS bidquote, CAST(( quote_depth_snapshot -> '$.quotes' ->> '$.midquote' ) AS REAL) AS midquote, ( quote_depth_snapshot -> '$.metadata' ->> '$.exchnote' ) AS exchange, ( quote_depth_snapshot -> '$.metadata' ->> '$.timetrack' ) AS timetrack FROM marketdata ORDER BY marketdatanode DESC NULLS FIRST LIMIT 1 ) SELECT exchange, timetrack, ( ( askquote - bidquote ) / midquote * 100 ) AS spread_percentage FROM snapshot;
