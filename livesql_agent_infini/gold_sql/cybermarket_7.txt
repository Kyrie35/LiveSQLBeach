WITH investigations AS ( SELECT i.investregistry, i.lawinterest, i.regrisklvl, i.compliancescore, i.notescount, ( CASE i.lawinterest WHEN 'Low' THEN 1 WHEN 'Medium' THEN 2 WHEN 'High' THEN 3 WHEN 'Unknown' THEN 2 END * 30 + CASE i.regrisklvl WHEN 'Low' THEN 1 WHEN 'Medium' THEN 2 WHEN 'High' THEN 3 WHEN 'Unknown' THEN 2 END * 20 + r.fraudprob * 100 - i.compliancescore * 0.5 + i.notescount * 2 ) AS investigation_priority_score, r.riskregistry FROM investigation AS i JOIN riskanalysis AS r ON i.riskref = r.riskregistry ) SELECT i.investregistry, i.investigation_priority_score, i.lawinterest, t.txregistry FROM investigations AS i JOIN riskanalysis AS r ON i.riskregistry = r.riskregistry JOIN transactions AS t ON r.txref = t.txregistry JOIN markets AS m ON t.mktref = m.mktregistry WHERE i.investigation_priority_score > 200 AND i.lawinterest = 'High' ORDER BY i.investigation_priority_score DESC NULLS FIRST;
