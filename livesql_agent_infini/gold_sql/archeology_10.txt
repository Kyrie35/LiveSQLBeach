WITH resource_metrics AS ( SELECT sp.flowregistry, sp.equipref, sp.flowhrs, sp.proccpu, sp.procgpu, sp.remaingb, sm.facetfaces, sm.facetverts, sp.flowhrs * (sp.proccpu + sp.procgpu) / 2.0 / (sp.remaingb * 10 * LOG(sm.facetverts + 10000)) as pru FROM scanprocessing sp LEFT JOIN scanmesh sm ON sp.zoneref = sm.zoneref ), workflow_analysis AS ( SELECT rm.equipref, rm.pru, ROUND(AVG(rm.flowhrs) OVER (PARTITION BY rm.equipref), 2) as avg_processing_hours, DENSE_RANK() OVER (ORDER BY rm.pru) as efficiency_rank, JSON_OBJECT( 'cpu_usage', rm.proccpu, 'gpu_usage', rm.procgpu, 'processing_hours', rm.flowhrs ) as resource_details FROM resource_metrics rm ) SELECT equipref, ROUND(pru, 2) as processing_resource_utilization, avg_processing_hours, efficiency_rank, CASE WHEN pru < 5.0 THEN 'Optimized' WHEN pru < 10.0 THEN 'Acceptable' ELSE 'Needs Optimization' END as workflow_status, resource_details FROM workflow_analysis ORDER BY pru ASC;
