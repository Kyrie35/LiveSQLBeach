WITH VendorRisk AS ( SELECT vm.VendorTrace, CASE WHEN vm.VendSecRate = 'A' THEN 4 WHEN vm.VendSecRate = 'B' THEN 3 WHEN vm.VendSecRate = 'C' THEN 2 ELSE 1 END * CASE WHEN vm.ContrState = 'Active' THEN 1 ELSE 0.5 END AS vri FROM VendorManagement AS vm ), TieredVendors AS ( SELECT CASE WHEN vri < 2 THEN 'High Risk' WHEN vri < 3 THEN 'Medium Risk' ELSE 'Low Risk' END AS risk_tier FROM VendorRisk ) SELECT risk_tier, COUNT(*) AS vendor_count, ROUND(COUNT(*) * 100.0 / SUM(COUNT(*)) OVER (), 2) AS percentage FROM TieredVendors GROUP BY risk_tier ORDER BY vendor_count DESC NULLS FIRST LIMIT 100;
