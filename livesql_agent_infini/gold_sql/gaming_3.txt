WITH rgb_analysis AS ( SELECT r.rgbregistry, d.makername, d.modnum, t.devscope, r.rgbbright, r.rgbcoloracc, r.rgbzones, r.rgbmodes, r.rgbcolors, ( r.rgbbright / 100.0 ) * ( r.rgbcoloracc / 10.0 ) * ( 0.5 + r.rgbzones / 20.0 ) AS riq FROM rgb AS r JOIN mechanical AS m ON r.rgbmechref = m.mechregistry JOIN deviceidentity AS d ON m.mechdevref = d.devregistry JOIN testsessions AS t ON d.devsessionref = t.sessionregistry ) SELECT makername AS manufacturer, modnum AS model, devscope AS device_type, rgbbright AS rgb_brightness, ROUND(rgbcoloracc, 2) AS rgb_color_accuracy, rgbzones AS rgb_lighting_zones, rgbmodes AS rgb_modes, rgbcolors AS supported_colors, ROUND(riq, 2) AS rgb_implementation_quality, CASE WHEN riq > 8.0 THEN 'Premium RGB Implementation' WHEN riq BETWEEN 6.0 AND 8.0 THEN 'High-Quality RGB' WHEN riq BETWEEN 3.0 AND 5.9 THEN 'Standard RGB' ELSE 'Basic RGB' END AS rgb_quality_category FROM rgb_analysis ORDER BY riq DESC NULLS FIRST;
