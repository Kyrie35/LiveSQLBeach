SELECT p.solmodref AS panel_id, pa.tempcoef AS temperature_coefficient, env.celltempc AS cell_temperature, CAST(( p.efficiency_profile -> '$.current_efficiency' ->> '$.curreffpct' ) AS REAL) AS current_efficiency, CAST(( env.irradiance_conditions -> '$.irradiance_types' ->> '$[3]' ) AS REAL) AS poa_irradiance, ROUND( ( CAST(( p.efficiency_profile -> '$.current_efficiency' ->> '$.curreffpct' ) AS REAL) * ( 1 + pa.tempcoef * ( 25 - env.celltempc ) / 100 ) * ( 1000 / ( CAST(( env.irradiance_conditions -> '$.irradiance_types' ->> '$[3]' ) AS REAL) ) ) ), 2 ) AS WCE FROM performance AS p INNER JOIN panel AS pa ON p.solmodref = pa.panemark INNER JOIN environment AS env ON env.arearegistry = pa.hubregistry;
