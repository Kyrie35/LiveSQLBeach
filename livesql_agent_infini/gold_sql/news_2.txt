WITH article_metrics AS ( SELECT artkey, catlabel, (qualscore + freshscore + sentscore - contrscore) / 3.0 AS aqi, CASE difflevel WHEN 'Basic' THEN 1.0 WHEN 'Intermediate' THEN 1.5 WHEN 'Advanced' THEN 2.0 ELSE 1.2 END AS diff_w, readsec, wordlen FROM articles ), article_ars AS ( SELECT artkey, catlabel, aqi, (readsec * LOG(wordlen)) / diff_w AS ars FROM article_metrics ) SELECT catlabel, json_group_array(json_object('artkey', artkey, 'catlabel', catlabel, 'aqi', aqi, 'ars', ars)) AS article_data FROM article_ars GROUP BY catlabel;
