WITH inefficient_robots AS ( SELECT rd.botdetreg AS robot_id, op.apptypeval AS application_type, rd.modelseriesval AS model_series, SUM(ps.energyusekwhval) / MAX(op.totopshrval) AS eer FROM robot_details AS rd JOIN operation AS op ON rd.botdetreg = op.operbotdetref JOIN performance_and_safety AS ps ON rd.botdetreg = ps.effectivenessrobot GROUP BY rd.botdetreg, op.apptypeval, rd.modelseriesval HAVING SUM(ps.energyusekwhval) / MAX(op.totopshrval) > 0.01 AND MAX(op.totopshrval) > 1000 ), eer_ranks AS ( SELECT application_type, robot_id, eer, PERCENT_RANK() OVER (PARTITION BY application_type ORDER BY eer DESC NULLS FIRST) AS eer_rank FROM inefficient_robots ) SELECT application_type, AVG(eer) AS average_eer, MAX(eer_rank) AS worst_eer_rank FROM eer_ranks GROUP BY application_type ORDER BY average_eer DESC NULLS FIRST LIMIT 100;
