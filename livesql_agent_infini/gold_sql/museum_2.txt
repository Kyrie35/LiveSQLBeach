WITH erf_calc AS ( SELECT a.artregistry, a.artname, AVG(CASE WHEN lower(trim(s.envsensitivity))='high' THEN 10 WHEN lower(trim(s.envsensitivity))='medium' THEN 5 WHEN lower(trim(s.envsensitivity))='low' THEN 1 END) AS ERF FROM artifactscore a LEFT JOIN sensitivitydata s ON s.artref = a.artregistry GROUP BY a.artregistry, a.artname ), high_sens AS ( SELECT t.artref, json_group_array(json_object('type', t.k, 'value', t.v)) AS high_sensitivities FROM ( SELECT s.artref, 'EnvSensitivity' AS k, s.envsensitivity AS v FROM sensitivitydata s WHERE lower(trim(s.envsensitivity))='high' UNION ALL SELECT s.artref, 'LightSensitivity', s.lightsensitivity FROM sensitivitydata s WHERE lower(trim(s.lightsensitivity))='high' UNION ALL SELECT s.artref, 'TempSensitivity', s.tempsensitivity FROM sensitivitydata s WHERE lower(trim(s.tempsensitivity))='high' UNION ALL SELECT s.artref, 'HumiditySensitivity', s.humiditysensitivity FROM sensitivitydata s WHERE lower(trim(s.humiditysensitivity))='high' UNION ALL SELECT s.artref, 'VibraSensitivity', s.vibrasensitivity FROM sensitivitydata s WHERE lower(trim(s.vibrasensitivity))='high' UNION ALL SELECT s.artref, 'PollutantSensitivity', s.pollutantsensitivity FROM sensitivitydata s WHERE lower(trim(s.pollutantsensitivity))='high' UNION ALL SELECT s.artref, 'PestSensitivity', s.pestsensitivity FROM sensitivitydata s WHERE lower(trim(s.pestsensitivity))='high' UNION ALL SELECT s.artref, 'HandleSensitivity', s.handlesensitivity FROM sensitivitydata s WHERE lower(trim(s.handlesensitivity))='high' UNION ALL SELECT s.artref, 'TransportSensitivity', s.transportsensitivity FROM sensitivitydata s WHERE lower(trim(s.transportsensitivity))='high' UNION ALL SELECT s.artref, 'DisplaySensitivity', s.displaysensitivity FROM sensitivitydata s WHERE lower(trim(s.displaysensitivity))='high' UNION ALL SELECT s.artref, 'StorageSensitivity', s.storagesensitivity FROM sensitivitydata s WHERE lower(trim(s.storagesensitivity))='high' ) t GROUP BY t.artref ), threshold AS ( SELECT MAX(7, 0) AS min_erf ) SELECT ec.artregistry, ec.artname, ROUND(ec.ERF,2) AS ERF, COALESCE(hs.high_sensitivities,'[]') AS high_sensitivities FROM erf_calc ec LEFT JOIN high_sens hs ON hs.artref = ec.artregistry, threshold th WHERE ec.ERF > th.min_erf ORDER BY ERF DESC LIMIT 100;
