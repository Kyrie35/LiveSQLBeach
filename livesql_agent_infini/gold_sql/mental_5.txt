WITH StablePatients AS ( SELECT asd.asdkey, ab.patownerref as patkey FROM assessmentsocialanddiagnosis asd JOIN assessmentbasics ab ON asd.asdkey = ab.abkey WHERE asd.recstatus = 'Stable' ), PatientAvgMissedAppts AS ( SELECT patref AS patkey, ROUND(AVG(missappt)) AS avg_missappt FROM encounters WHERE missappt IS NOT NULL GROUP BY patref ), PatientSocialSupport AS ( SELECT ab.patownerref AS patkey, asd.asdkey, (CASE asd.socsup WHEN 'Strong' THEN 3 WHEN 'Moderate' THEN 2 WHEN 'Limited' THEN 1 ELSE 0 END + CASE asd.relqual WHEN 'Good' THEN 3 WHEN 'Fair' THEN 2 WHEN 'Poor' THEN 1 WHEN 'Conflicted' THEN 0 ELSE 0 END) AS sse_score FROM assessmentsocialanddiagnosis asd JOIN assessmentbasics ab ON asd.asdkey = ab.abkey ), LatestPatientSSE AS ( SELECT sp.patkey, MAX(pss.sse_score) AS sse_score FROM StablePatients sp JOIN PatientSocialSupport pss ON sp.patkey = pss.patkey JOIN encounters e ON pss.asdkey = e.abref AND e.patref = sp.patkey GROUP BY sp.patkey ) SELECT sp.patkey, pama.avg_missappt, lsse.sse_score FROM StablePatients sp JOIN PatientAvgMissedAppts pama ON sp.patkey = pama.patkey JOIN LatestPatientSSE lsse ON sp.patkey = lsse.patkey WHERE pama.avg_missappt > 2 OR lsse.sse_score < 3;
