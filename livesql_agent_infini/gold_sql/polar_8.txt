SELECT commregistry, antennastatus, ROUND(JSON_EXTRACT(signalmetrics, '$.radio_strength_dbm'), 2) AS radio_strength, ROUND(JSON_EXTRACT(signalmetrics, '$.latency_ms'), 2) AS latency_ms, ROUND( CASE WHEN antennastatus = 'Error' THEN 0 WHEN antennastatus = 'Warning' THEN 5 WHEN antennastatus = 'Normal' THEN 10 ELSE 0 END * (1 - (JSON_EXTRACT(signalmetrics, '$.latency_ms')/1000)), 2 ) AS CRI, ROUND( CASE WHEN JSON_EXTRACT(signalmetrics, '$.latency_ms') IS NOT NULL THEN (CASE WHEN antennastatus = 'Error' THEN 0 WHEN antennastatus = 'Warning' THEN 5 WHEN antennastatus = 'Normal' THEN 10 ELSE 0 END) * (1 + 0.2 * (JSON_EXTRACT(signalmetrics, '$.radio_strength_dbm')/100)) * (1 - 0.01 * (1000 - JSON_EXTRACT(signalmetrics, '$.latency_ms'))) ELSE NULL END, 2 ) AS BSCSI FROM communication;
