WITH TraderBaseMetrics AS ( SELECT t.tradereg, t.voldaily / IIF(t.acctbal = 0, NULL, t.acctbal) AS dtr, CAST(REPLACE(REPLACE(json_extract(t.trading_performance, '$.risklevel.levratio'), '[', ''), ']', '') AS REAL) AS tle, json_extract(t.trading_performance, '$.risklevel.risklevel') AS risk_level_text FROM trader t ), HighRiskTraders AS ( SELECT tradereg FROM TraderBaseMetrics WHERE tle > 5.0 AND risk_level_text = 'Aggressive' OR dtr > 0.5 ), ManipulationPatternTraders AS ( SELECT DISTINCT tr.trdref AS tradereg FROM transactionrecord tr LEFT JOIN ( SELECT transreg, modfreq / IIF(1 - cancelpct = 0, NULL, 1 - cancelpct) AS omi FROM transactionrecord WHERE cancelpct < 1 ) AS omi_calc ON tr.transreg = omi_calc.transreg WHERE json_extract(tr.risk_indicators, '$.layerind') = 'Confirmed' OR ( CAST(REPLACE(REPLACE(json_extract(tr.risk_indicators, '$.spoofprob'), '[', ''), ']', '') AS REAL) > 0.75 AND IFNULL(omi_calc.omi, 0) > 1.0 ) ), HighRiskManipulators AS ( SELECT hrt.tradereg FROM HighRiskTraders hrt INTERSECT SELECT mpt.tradereg FROM ManipulationPatternTraders mpt ), AdvancedBehaviorMetrics AS ( SELECT tr.trdref AS tradereg, AVG(ABS(ab.patsim - ab.peercorr)) AS avg_pas, AVG(ab.patsim) AS avg_patsim FROM advancedbehavior ab JOIN transactionrecord tr ON ab.translink = tr.transreg GROUP BY tr.trdref ) SELECT ROUND(AVG(abm.avg_pas / MAX(0.01, abm.avg_patsim)), 2) AS average_updr_for_high_risk_manipulators FROM AdvancedBehaviorMetrics abm WHERE abm.tradereg IN (SELECT tradereg FROM HighRiskManipulators) LIMIT 100;
