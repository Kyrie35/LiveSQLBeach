SELECT d.makername AS manufacturer, d.modnum AS model, d.dpires AS dpi_resolution, t.pollratehz AS polling_rate, ROUND(spi.sensor_performance, 2) AS spi_score, ROUND(ci.comfort_index, 2) AS ci_score, CASE WHEN spi.sensor_performance > 7.5 AND d.dpires >= 16000 AND t.pollratehz >= 1000 AND ci.comfort_index > 8.0 THEN 'Yes' ELSE 'No' END AS is_premium_gaming_mouse FROM deviceidentity AS d JOIN testsessions AS t ON d.devsessionref = t.sessionregistry JOIN ( SELECT devregistry, ( dpires / 1000.0 ) * ( 1 - mcresptime / 10.0 ) * 10 AS sensor_performance FROM deviceidentity ) AS spi ON d.devregistry = spi.devregistry JOIN ( SELECT m.mechdevref, ( m.ergorate / 10.0 ) * ( 1 + ( CASE WHEN m.wristflag THEN 1 ELSE 0 END ) / 5.0 ) * ( 1 - ABS(m.palmangle - 15) / 45.0 ) * 10 AS comfort_index FROM mechanical AS m ) AS ci ON d.devregistry = ci.mechdevref WHERE t.devscope = 'Mouse' ORDER BY spi_score DESC NULLS FIRST, ci_score DESC NULLS FIRST;
