WITH spi_cte AS ( SELECT devregistry, ( dpires / 1000.0 ) * ( 1 - mcresptime / 10.0 ) * 10 AS spi FROM deviceidentity ), irs_cte AS ( SELECT d.devregistry, ( t.pollratehz / 100.0 ) * ( 1 - ( t.inplagms + t.resptimems ) / 30.0 ) * 10 AS irs FROM deviceidentity AS d JOIN testsessions AS t ON d.devsessionref = t.sessionregistry ), spr_cte AS ( SELECT m.mechdevref, ( LOG(10, m.swtchdur) / 7.0 ) * m.swtchcons * ( 1 - m.keychatter / 2.0 ) AS spr FROM mechanical AS m ), rai_cte AS ( SELECT i.interactdevref, ( ( i.joyprec + i.dpadacc ) / 20.0 ) * ( 1 - t.inplagms / 10.0 ) * 10 AS rai FROM interactionandcontrol AS i JOIN deviceidentity AS d ON i.interactdevref = d.devregistry JOIN testsessions AS t ON d.devsessionref = t.sessionregistry ), cgpi_cte AS ( SELECT d.devregistry, ( irs.irs * 0.4 ) + ( spi.spi * 0.3 ) + ( spr.spr * 0.2 ) + ( rai.rai * 0.1 ) AS cgpi, d.profcount FROM deviceidentity AS d JOIN spi_cte AS spi ON d.devregistry = spi.devregistry JOIN irs_cte AS irs ON d.devregistry = irs.devregistry JOIN spr_cte AS spr ON d.devregistry = spr.mechdevref JOIN rai_cte AS rai ON d.devregistry = rai.interactdevref ) SELECT c.devregistry, ROUND(c.cgpi, 2) AS cgpi, c.profcount, ROUND(spi.spi, 2) AS spi, ROUND(irs.irs, 2) AS irs, ROUND( ( c.cgpi / 10.0 ) * ( 1 + c.profcount / 5.0 ) * ( ( spi.spi + irs.irs ) / 15.0 ), 2 ) AS par FROM cgpi_cte AS c JOIN spi_cte AS spi ON c.devregistry = spi.devregistry JOIN irs_cte AS irs ON c.devregistry = irs.devregistry WHERE ( c.cgpi / 10.0 ) * ( 1 + c.profcount / 5.0 ) * ( ( spi.spi + irs.irs ) / 15.0 ) > 8.5 ORDER BY par DESC NULLS FIRST;
