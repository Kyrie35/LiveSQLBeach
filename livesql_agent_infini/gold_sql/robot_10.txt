WITH program_metrics AS ( SELECT o.operbotdetref, o.currprogval, COUNT(*) as execution_count, AVG(o.cycletimesecval) as avg_cycle_time, SUM(o.progcyclecount) as total_cycles, SUM(o.progcyclecount)*1.0/NULLIF(SUM(o.cycletimesecval), 0) as program_oce FROM operation o GROUP BY o.operbotdetref, o.currprogval ), robot_efficiency AS ( SELECT pm.operbotdetref, COUNT(DISTINCT pm.currprogval) as program_count, AVG(pm.avg_cycle_time) as overall_cycle_time, SUM(pm.total_cycles) as total_program_cycles, SUM(pm.total_cycles)*1.0/NULLIF(SUM(pm.total_cycles * pm.avg_cycle_time), 0) as robot_oce, GROUP_CONCAT(DISTINCT pm.currprogval) as programs_used FROM program_metrics pm GROUP BY pm.operbotdetref ), program_ranking AS ( SELECT pm.currprogval, AVG(pm.program_oce) as avg_program_oce, COUNT(DISTINCT pm.operbotdetref) as robot_count, RANK() OVER (ORDER BY AVG(pm.program_oce) DESC) as efficiency_rank FROM program_metrics pm GROUP BY pm.currprogval ) SELECT rd.modelseriesval, COUNT(DISTINCT re.operbotdetref) as robot_count, ROUND(AVG(re.robot_oce), 2) as avg_oce, ROUND(AVG(re.program_count), 2) as avg_programs_per_robot, ROUND(AVG(re.total_program_cycles), 0) as avg_total_cycles, json_object( 'most_efficient_program', ( SELECT currprogval FROM program_ranking WHERE efficiency_rank = 1 LIMIT 1 ), 'avg_program_efficiency', ( SELECT ROUND(AVG(avg_program_oce), 2) FROM program_ranking ) ) as efficiency_metrics, SUM(CASE WHEN re.robot_oce < 100 AND re.total_program_cycles > 500000 THEN 1 ELSE 0 END) as low_efficiency_robots FROM robot_details rd JOIN robot_efficiency re ON rd.botdetreg = re.operbotdetref GROUP BY rd.modelseriesval HAVING COUNT(*) >= 5 ORDER BY avg_oce DESC LIMIT 50;
