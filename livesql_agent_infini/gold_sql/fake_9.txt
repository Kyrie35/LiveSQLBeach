WITH account_activity AS ( SELECT a.accindex, COUNT(sb.sessref) AS session_count, SUM(cb.postfreq) AS total_post_frequency FROM account AS a JOIN profile AS p ON p.profaccref = a.accindex JOIN sessionbehavior AS sb ON sb.sessprofref = p.profkey JOIN contentbehavior AS cb ON cb.cntsessref = sb.sessref GROUP BY a.accindex ), srs_calc AS ( SELECT a.accindex, COALESCE( MAX( 0.4 * CAST(( s.detection_score_profile -> '$.overall' ->> '$.riskval' ) AS REAL) + 0.3 * ( 1 - m.trustval ) + 0.3 * m.impactval ), 0 ) AS srs, COALESCE(MAX(t.ipcountrynum), 0) AS ipcountrynum FROM account AS a JOIN profile AS p ON p.profaccref = a.accindex JOIN sessionbehavior AS sb ON sb.sessprofref = p.profkey JOIN contentbehavior AS cb ON cb.cntsessref = sb.sessref LEFT JOIN moderationaction AS m ON m.macntref = cb.cntref LEFT JOIN securitydetection AS s ON s.secdetkey = m.masedetref LEFT JOIN technicalinfo AS t ON t.technetref = ( SELECT netkey FROM networkmetrics AS nm WHERE nm.netsessref = sb.sessref LIMIT 1 ) GROUP BY a.accindex ) SELECT aa.accindex, aa.session_count, aa.total_post_frequency, 1 AS platform_count, ( sc.srs * ( 1 + 0.2 * sc.ipcountrynum ) ) AS cpri FROM account_activity AS aa JOIN srs_calc AS sc ON aa.accindex = sc.accindex WHERE aa.session_count > 1000 OR aa.total_post_frequency > 50 ORDER BY cpri DESC NULLS FIRST LIMIT 10;
