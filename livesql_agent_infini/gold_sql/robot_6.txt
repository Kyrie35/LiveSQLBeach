WITH tool_metrics AS ( SELECT effectivenessrobot, AVG(toolwearpct) AS avg_wear, SUM(toolwearpct) / NULLIF(SUM(tc.progcyclecount), 0) AS twr FROM performance_and_safety AS ps JOIN operation AS tc ON ps.effectivenessrobot = tc.operbotdetref GROUP BY effectivenessrobot ), cycle_metrics AS ( SELECT operbotdetref, SUM(progcyclecount) AS tpc, AVG(cycletimesecval) AS avg_cycle_time FROM operation GROUP BY operbotdetref ) SELECT rd.botdetreg, tm.avg_wear, tm.twr, cm.tpc, cm.avg_cycle_time, CASE WHEN tm.twr > 0.001 AND cm.tpc > 10000 THEN 'URGENT' WHEN tm.twr > 0.0005 OR tm.avg_wear > 75 THEN 'WARNING' ELSE 'NORMAL' END AS tool_status FROM robot_details AS rd JOIN tool_metrics AS tm ON rd.botdetreg = tm.effectivenessrobot JOIN cycle_metrics AS cm ON rd.botdetreg = cm.operbotdetref ORDER BY tm.twr DESC NULLS FIRST, cm.tpc DESC NULLS FIRST LIMIT 100;
