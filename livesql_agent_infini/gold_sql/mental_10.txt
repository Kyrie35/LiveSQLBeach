WITH FacilityPFIS AS ( SELECT e.facid, AVG( CASE asd.funcimp WHEN 'Severe' THEN 3 WHEN 'Moderate' THEN 2 WHEN 'Mild' THEN 1 ELSE 0 END ) AS pfis FROM assessmentsocialanddiagnosis AS asd JOIN assessmentbasics AS ab ON asd.asdkey = ab.abkey JOIN encounters AS e ON ab.abkey = e.abref WHERE NOT e.facid IS NULL AND NOT asd.funcimp IS NULL GROUP BY e.facid ), FacilityFRAI AS ( SELECT fackey AS facid, CASE ( support_and_resources ->> '$.community_resources' ) WHEN 'Comprehensive' THEN 3 WHEN 'Adequate' THEN 2 WHEN 'Limited' THEN 1 ELSE 0 END AS frai FROM facilities WHERE NOT support_and_resources ->> '$.community_resources' IS NULL ), FacilityRDD AS ( SELECT pfis.facid, ROUND(( pfis.pfis - ff.frai ), 2) AS rdd FROM FacilityPFIS AS pfis JOIN FacilityFRAI AS ff ON pfis.facid = ff.facid ), FacilityTES AS ( SELECT e.facid, AVG( CASE ( tb.therapy_details ->> '$.engagement' ) WHEN 'High' THEN 3 WHEN 'Medium' THEN 2 WHEN 'Low' THEN 1 WHEN 'Non-compliant' THEN 0 ELSE 0 END ) AS avg_tes FROM treatmentbasics AS tb JOIN encounters AS e ON tb.encref = e.enckey WHERE NOT tb.therapy_details IS NULL AND NOT e.facid IS NULL GROUP BY e.facid ), FacilityTAR AS ( SELECT e.facid, CASE WHEN COUNT(tro.txoutkey) > 0 THEN CAST(COUNT(*) FILTER(WHERE tro.txadh IN ('High', 'Medium')) AS REAL) / COUNT(tro.txoutkey) ELSE 0 END AS tar FROM treatmentoutcomes AS tro JOIN treatmentbasics AS tb ON tro.txref = tb.txkey JOIN encounters AS e ON tb.encref = e.enckey WHERE NOT e.facid IS NULL GROUP BY e.facid ), FacilityEAS AS ( SELECT tes.facid, ( tes.avg_tes + ( tar.tar * 3.0 ) ) / 2.0 AS eas FROM FacilityTES AS tes JOIN FacilityTAR AS tar ON tes.facid = tar.facid ), FacilityMAR AS ( SELECT facid, AVG(COALESCE(missappt, 0)) AS mar FROM encounters WHERE NOT facid IS NULL GROUP BY facid ), AttritionRiskFacilities AS ( SELECT feas.facid FROM FacilityEAS AS feas JOIN FacilityMAR AS fmar ON feas.facid = fmar.facid WHERE feas.eas < 1.5 AND fmar.mar > 2.5 ) SELECT frdd.facid, frdd.rdd FROM FacilityRDD AS frdd WHERE frdd.rdd > 1.0 AND frdd.facid IN ( SELECT facid FROM AttritionRiskFacilities ) ORDER BY frdd.rdd DESC NULLS FIRST LIMIT 100;
