WITH TraderMetrics AS ( SELECT t.tradereg, t.tradekind, t.acctdays, t.voldaily / NULLIF(t.acctbal, 0) AS dtr, CAST(REPLACE(REPLACE(json_extract(t.trading_performance, '$.risklevel.levratio'), '[', ''), ']', '') AS REAL) AS tle FROM trader t ), TransactionAvgMetrics AS ( SELECT trdref, AVG(modfreq / NULLIF(1 - cancelpct, 0)) AS avg_omi FROM transactionrecord WHERE cancelpct < 1 GROUP BY trdref ), ComplianceMetrics AS ( SELECT tr.trdref, AVG(cc.prevviol / CAST(MAX(1, tm.acctdays / 365.0) AS REAL)) AS avg_crs, AVG(CASE cc.comprate WHEN 'A' THEN 1 WHEN 'B' THEN 2 WHEN 'C' THEN 3 WHEN 'D' THEN 4 ELSE 1 END) AS avg_comprate_value FROM compliancecase cc JOIN transactionrecord tr ON cc.transref = tr.transreg JOIN TraderMetrics tm ON tr.trdref = tm.tradereg GROUP BY tr.trdref ), CombinedScores AS ( SELECT tm.tradereg, tm.tradekind, (tm.dtr * tm.tle * COALESCE(tam.avg_omi, 0)) AS ati, 1.0 / (1.0 + COALESCE(cm.avg_crs, 0) * COALESCE(cm.avg_comprate_value, 1)) AS chs FROM TraderMetrics tm LEFT JOIN TransactionAvgMetrics tam ON tm.tradereg = tam.trdref LEFT JOIN ComplianceMetrics cm ON tm.tradereg = cm.trdref ) SELECT tradekind, ROUND(AVG(ati), 2) AS average_ati, ROUND(AVG(chs), 2) AS average_chs FROM CombinedScores GROUP BY tradekind ORDER BY average_ati DESC LIMIT 100;
