SELECT r.churnflag, COUNT(*) AS fan_count, ROUND(CAST(AVG(CURRENT_DATE - p.lastlogdt) AS REAL), 2) AS avg_days_since_login, ROUND(CAST(AVG(e.engrate) AS REAL), 2) AS avg_engagement_rate, ROUND(CAST(AVG(p.intconsist) AS REAL), 2) AS avg_interaction_consistency, ROUND( CAST(AVG( ( 1 - p.intconsist ) * 2 + ( CURRENT_DATE - p.lastlogdt ) / 30.0 * 0.5 + CASE WHEN r.churnflag = 'None' THEN 0 WHEN r.churnflag = 'Low' THEN 1 WHEN r.churnflag = 'Medium' THEN 2 WHEN r.churnflag = 'High' THEN 3 END ) AS REAL), 2 ) AS avg_retention_risk_factor FROM fans AS f JOIN membershipandspending AS ms ON ms.memberfanpivot = f.userregistry JOIN engagement AS e ON e.engagememberpivot = ms.memberreg JOIN retentionandinfluence AS r ON r.retainengagepivot = e.engagereg JOIN preferencesandsettings AS p ON p.preferencesmemberpivot = ms.memberreg GROUP BY r.churnflag;
