WITH operation_metrics AS ( SELECT o.opsregistry, d.affectedarea, t.deliverysuccessrate, h.invaccpct, ( t.deliverysuccessrate / 100.0 ) * ( 1 - t.avgdeliveryhours / 24.0 ) * ( 1 + t.distributionpoints / 10.0 ) AS oei FROM operations AS o JOIN disasterevents AS d ON o.opsdistref = d.distregistry JOIN distributionhubs AS h ON o.opshubref = h.hubregistry JOIN transportation AS t ON o.opsdistref = t.transportdistref ) SELECT opsregistry, ROUND(CAST(oei AS REAL), 2) AS operational_efficiency_index, CASE WHEN deliverysuccessrate > 90 AND invaccpct > 95 AND oei > 3 THEN 'Yes' ELSE 'No' END AS operational_excellence FROM operation_metrics ORDER BY oei DESC NULLS FIRST;
