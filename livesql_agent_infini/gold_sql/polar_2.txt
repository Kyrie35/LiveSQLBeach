WITH structural_safety AS ( SELECT ws.weatherlocref, ROUND( CAST(( ( 100 - ws.structuralloadpercent ) / 100 * CASE WHEN ws.snowloadkgm2 > 100 OR ws.windspeedms > 20 THEN 0.5 WHEN ws.snowloadkgm2 > 50 OR ws.windspeedms > 10 THEN 0.8 ELSE 1.0 END ) AS REAL), 2 ) AS ssf FROM weatherandstructure AS ws ) SELECT e.equipmentcode, e.equipmenttype, ce.heaterstatus, tswg.insulationstatus, ls.emergencylightstatus, ss.ssf, CASE WHEN ss.ssf > 0.7 AND ce.heaterstatus <> 'Off' AND tswg.insulationstatus <> 'Poor' AND ( ls.emergencylightstatus = 'On' OR ls.emergencylightstatus = 'Testing' ) THEN 'Extreme Weather Ready' ELSE 'Not Ready' END AS ewr_status FROM equipment AS e JOIN location AS l ON e.equipmentcode = l.loceqref LEFT JOIN cabinenvironment AS ce ON e.equipmentcode = ce.cabineqref LEFT JOIN lightingandsafety AS ls ON e.equipmentcode = ls.lighteqref LEFT JOIN thermalsolarwindandgrid AS tswg ON e.equipmentcode = tswg.thermaleqref LEFT JOIN structural_safety AS ss ON l.locationregistry = ss.weatherlocref WHERE NOT ss.ssf IS NULL;
