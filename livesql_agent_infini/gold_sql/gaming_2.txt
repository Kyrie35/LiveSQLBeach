WITH ber_calc AS ( SELECT d.devregistry, d.makername, d.modnum, t.battlifeh, t.battcapmah, d.pwractmw, ( t.battlifeh * t.battcapmah ) / ( d.pwractmw * 10.0 ) AS ber FROM deviceidentity AS d JOIN testsessions AS t ON d.devsessionref = t.sessionregistry WHERE d.conntype LIKE '%Wireless%' OR d.conntype LIKE '%Bluetooth%' ) SELECT makername AS manufacturer, modnum AS model, battlifeh AS battery_life_hours, battcapmah AS battery_capacity_mah, pwractmw AS active_power_draw_mw, ROUND(ber, 2) AS battery_efficiency_ratio, CASE WHEN ber > 7.5 THEN 'Excellent Efficiency' WHEN ber BETWEEN 5.0 AND 7.5 THEN 'Good Efficiency' WHEN ber BETWEEN 2.5 AND 4.9 THEN 'Average Efficiency' ELSE 'Poor Efficiency' END AS efficiency_category, PERCENT_RANK() OVER (PARTITION BY makername ORDER BY ber NULLS LAST) AS efficiency_percentile FROM ber_calc ORDER BY ber DESC NULLS FIRST;
