WITH PatientDiagnosis AS ( SELECT patownerref AS patkey, primdx FROM ( SELECT ab.patownerref, asd.primdx, ROW_NUMBER() OVER (PARTITION BY ab.patownerref ORDER BY e.timemark DESC) as rn FROM assessmentbasics ab JOIN assessmentsocialanddiagnosis asd ON ab.abkey = asd.asdkey JOIN encounters e ON ab.abkey = e.abref WHERE asd.primdx IS NOT NULL ) ranked WHERE rn = 1 ), PatientCrisisInterventions AS ( SELECT e.patref AS patkey, SUM(COALESCE(tb.crisisint, 0)) AS total_crisisint FROM treatmentbasics tb JOIN encounters e ON tb.encref = e.enckey GROUP BY e.patref ), PatientCount AS ( SELECT COUNT(DISTINCT patkey) AS total_patients FROM patients ) SELECT pd.primdx, COUNT(DISTINCT pci.patkey) AS num_patients_with_dx, ROUND(CAST(SUM(pci.total_crisisint) AS REAL) / pc.total_patients, 2) AS cif_per_diagnosis FROM PatientDiagnosis pd JOIN PatientCrisisInterventions pci ON pd.patkey = pci.patkey CROSS JOIN PatientCount pc GROUP BY pd.primdx, pc.total_patients ORDER BY cif_per_diagnosis DESC LIMIT 100;
