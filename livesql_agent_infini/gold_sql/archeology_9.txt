WITH feature_potential AS ( SELECT s.zoneref AS site_code, spc.totalpts, spc.clouddense, ss.aream2, CASE WHEN spc.clouddense > 0 AND ss.aream2 > 0 THEN spc.totalpts / ( spc.clouddense * ss.aream2 ) ELSE 0 END AS pcdr FROM sites AS si LEFT JOIN scans AS s ON si.zoneregistry = s.zoneref LEFT JOIN scanpointcloud AS spc ON s.arcref = spc.arcref LEFT JOIN scanspatial AS ss ON s.arcref = ss.arcref ), extraction_metrics AS ( SELECT fp.site_code, fp.pcdr, sf.traitcount, sf.articount, CASE WHEN fp.pcdr > 0 THEN ( sf.traitcount + sf.articount ) / ( fp.pcdr * SQRT(fp.clouddense) ) * 1000 ELSE 0 END AS fee FROM feature_potential AS fp JOIN scanfeatures AS sf ON fp.site_code = sf.zoneref ) SELECT site_code, ROUND(CAST(AVG(pcdr) AS REAL), 2) AS avg_pcdr, ROUND(CAST(AVG(fee) AS REAL), 2) AS avg_fee FROM extraction_metrics GROUP BY site_code ORDER BY avg_fee DESC NULLS FIRST;
