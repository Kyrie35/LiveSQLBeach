WITH wrmi_calc AS ( SELECT waw.watereqref, waw.waterlevelpercent * ( waw.waterqualityindex / 100 ) * ( 1 - waw.wastetanklevelpercent / 100 ) AS wrmi FROM waterandwaste AS waw ) SELECT l.stationname, l.locationtype, ROUND(CAST(AVG(waw.waterqualityindex) AS REAL), 2) AS avg_water_quality_index, ROUND(CAST(AVG(wc.wrmi) AS REAL), 2) AS avg_wrmi, COUNT(*) FILTER(WHERE wc.wrmi < 0.5) AS systems_needing_conservation, CASE WHEN AVG(waw.waterqualityindex) >= 91 THEN 'High-Quality' WHEN AVG(waw.waterqualityindex) >= 71 THEN 'Good' WHEN AVG(waw.waterqualityindex) >= 51 THEN 'Moderate' WHEN AVG(waw.waterqualityindex) >= 26 THEN 'Poor' ELSE 'Unsafe' END AS quality_classification, CASE WHEN AVG(wc.wrmi) < 0.5 THEN 'Conservation Needed' WHEN AVG(wc.wrmi) < 0.7 THEN 'Monitoring Advised' ELSE 'Sustainable Management' END AS management_status FROM waterandwaste AS waw JOIN equipment AS e ON waw.watereqref = e.equipmentcode JOIN location AS l ON e.equipmentcode = l.loceqref JOIN wrmi_calc AS wc ON waw.watereqref = wc.watereqref GROUP BY l.stationname, l.locationtype ORDER BY avg_wrmi DESC NULLS FIRST, avg_water_quality_index DESC NULLS FIRST;
