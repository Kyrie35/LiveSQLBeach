WITH bounce_analysis AS ( SELECT s.seshkey, s.userel, s.seshdur, s.seshviews, s.bncrate, s.ctrval, s.engscore, s.bncrate * ( 1 - s.ctrval / 100 ) AS adjusted_bounce_rate, u.typelabel, u.seglabel FROM sessions AS s JOIN users AS u ON s.userel = u.userkey WHERE NOT s.bncrate IS NULL AND NOT s.ctrval IS NULL ), percentile_analysis AS ( SELECT seshkey, userel, typelabel, seglabel, seshdur, seshviews, bncrate, ctrval, engscore, adjusted_bounce_rate, PERCENT_RANK() OVER (ORDER BY adjusted_bounce_rate NULLS LAST) AS bounce_percentile, PERCENT_RANK() OVER (ORDER BY ctrval DESC NULLS FIRST) AS ctr_percentile, PERCENT_RANK() OVER (ORDER BY engscore DESC NULLS FIRST) AS engagement_percentile FROM bounce_analysis ) SELECT typelabel AS user_type, seglabel AS user_segment, CASE WHEN bounce_percentile < 0.25 AND ctr_percentile < 0.25 THEN 'High Bounce, Low CTR' WHEN bounce_percentile < 0.25 AND ctr_percentile >= 0.75 THEN 'High Bounce, High CTR' WHEN bounce_percentile >= 0.75 AND ctr_percentile < 0.25 THEN 'Low Bounce, Low CTR' WHEN bounce_percentile >= 0.75 AND ctr_percentile >= 0.75 THEN 'Low Bounce, High CTR' ELSE 'Average Performance' END AS performance_segment, COUNT(*) AS session_count, ROUND(CAST(AVG(bncrate) AS REAL), 2) AS avg_bounce_rate, ROUND(CAST(AVG(adjusted_bounce_rate) AS REAL), 2) AS avg_adjusted_bounce_rate, ROUND(CAST(AVG(ctrval) AS REAL), 2) AS avg_ctr, ROUND(CAST(AVG(engscore) AS REAL), 2) AS avg_engagement_score, ROUND(CAST(AVG(seshdur) AS REAL), 2) AS avg_session_duration, ROUND(CAST(AVG(seshviews) AS REAL), 2) AS avg_page_views FROM percentile_analysis GROUP BY typelabel, seglabel, CASE WHEN bounce_percentile < 0.25 AND ctr_percentile < 0.25 THEN 'High Bounce, Low CTR' WHEN bounce_percentile < 0.25 AND ctr_percentile >= 0.75 THEN 'High Bounce, High CTR' WHEN bounce_percentile >= 0.75 AND ctr_percentile < 0.25 THEN 'Low Bounce, Low CTR' WHEN bounce_percentile >= 0.75 AND ctr_percentile >= 0.75 THEN 'Low Bounce, High CTR' ELSE 'Average Performance' END HAVING COUNT(*) > 10 ORDER BY avg_adjusted_bounce_rate NULLS LAST, avg_ctr DESC NULLS FIRST;
