WITH irs_calc AS ( SELECT d.devregistry, d.makername, d.modnum, ( t.pollratehz / 100.0 ) * ( 1 - ( t.inplagms + t.resptimems ) / 30.0 ) * 10 AS irs FROM deviceidentity AS d JOIN testsessions AS t ON d.devsessionref = t.sessionregistry WHERE t.devscope = 'Controller' ), controller_metrics AS ( SELECT i.interactdevref, i.joyprec, i.driftres, i.trigres, i.hapstr FROM interactionandcontrol AS i JOIN deviceidentity AS d ON i.interactdevref = d.devregistry JOIN testsessions AS t ON d.devsessionref = t.sessionregistry WHERE t.devscope = 'Controller' ) SELECT d.makername AS manufacturer, d.modnum AS model, ROUND(irs.irs, 2) AS input_responsiveness_score, c.joyprec AS joystick_precision, c.driftres AS drift_resistance, c.trigres AS trigger_resistance, c.hapstr AS haptic_strength, CASE WHEN irs.irs > 8.0 AND c.joyprec > 9.0 AND c.driftres > 9.5 AND c.trigres >= 5 AND c.hapstr > 8 THEN 'Yes' ELSE 'No' END AS meets_pro_standard FROM deviceidentity AS d JOIN irs_calc AS irs ON d.devregistry = irs.devregistry JOIN controller_metrics AS c ON d.devregistry = c.interactdevref ORDER BY meets_pro_standard DESC NULLS FIRST, input_responsiveness_score DESC NULLS FIRST;
