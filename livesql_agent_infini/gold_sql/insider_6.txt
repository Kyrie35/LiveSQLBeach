WITH HighIntensityInsiderInvestigations AS ( SELECT idet.invdetreg, idet.compref FROM investigationdetails AS idet JOIN sentimentandfundamentals AS sf ON idet.sentref = sf.sentreg WHERE sf.infoleaksc > 50.0 AND NOT sf.corpeventprx IS NULL AND sf.eventannotm IN ('Pre-market', 'Intraday') AND ( ( 0.6 * idet.behansc ) + ( 0.4 * idet.netansc ) ) > 70 ), EnforcementOutcomes AS ( SELECT hiii.invdetreg, ea.penimp FROM HighIntensityInsiderInvestigations AS hiii JOIN enforcementactions AS ea ON hiii.compref = ea.compref2 ) SELECT penimp, COUNT(*) AS frequency FROM EnforcementOutcomes GROUP BY penimp ORDER BY frequency DESC NULLS FIRST LIMIT 100;
