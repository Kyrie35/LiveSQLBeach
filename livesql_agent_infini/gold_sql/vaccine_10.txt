WITH HandlingMetrics AS (SELECT t.VehicleReg, t.TransMode, (1.0 - COALESCE(s.HandleEvents / 100.0, 0)) * (1.0 - COALESCE(s.CritEvents / 10.0, 0)) AS HQI, COALESCE((1.0 - COALESCE(s.TempDevCount / 100.0, 0)) * (1.0 - COALESCE(s.CritEvents / 10.0, 0)) * COALESCE(s.TempStabIdx, 1.0), 1.0) AS TSS FROM TransportInfo AS t INNER JOIN SensorData AS s ON t.VehicleReg = s.VehSenseRef) SELECT TransMode, ROUND(AVG(HQI), 3) AS avg_hqi, GROUP_CONCAT(CASE WHEN HQI > 0.9 AND TSS > 0.8 THEN VehicleReg END, ',') AS vehicles, SUM(CASE WHEN HQI > 0.9 AND TSS > 0.8 THEN 1 ELSE 0 END) AS stable_transports FROM HandlingMetrics GROUP BY TransMode HAVING AVG(HQI) > 0 ORDER BY avg_hqi DESC LIMIT 10;
