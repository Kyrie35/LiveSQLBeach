WITH StabilityMetrics AS ( SELECT c.containregistry, (1 - COALESCE(CAST(s.tempdevcount AS REAL) / 100.0, 0)) * (1 - COALESCE(CAST(s.critevents AS REAL) / 10.0, 0)) * COALESCE(s.tempstabidx, 1.0) AS tss, COALESCE(ABS(s.tempnowc - s.storetempc) / NULLIF(s.temptolc, 0) * s.tempdevcount, 0.0) AS tbs, s.tempdevcount, s.critevents, s.tempstabidx, CASE WHEN s.sensortrack IS NULL THEN 'No Sensor Data' ELSE 'Sensor Data Available' END AS sensor_data_status FROM container c LEFT JOIN sensordata s ON c.containregistry = s.containlink AND s.alerttime >= date((SELECT MAX(alerttime) FROM sensordata), '-1 year') ), RiskIndex AS ( SELECT sm.containregistry, COALESCE(MAX(sm.tss), 1.0) AS tss, COALESCE(MAX(sm.tbs), 0.0) AS max_tbs, (1 - COALESCE(MAX(sm.tss), 1.0)) * (1 - COALESCE(c.coolremainpct, 0) / 100.0) AS cri, MAX(sm.sensor_data_status) AS sensor_data_status, RANK() OVER (ORDER BY (1 - COALESCE(MAX(sm.tss), 1.0)) * (1 - COALESCE(c.coolremainpct, 0) / 100.0) DESC) AS risk_rank, json_object('TempDevCount', MAX(sm.tempdevcount), 'CritEvents', MAX(sm.critevents), 'TempStabIdx', MAX(sm.tempstabidx), 'CoolRemainPct', c.coolremainpct, 'TempBreachSeverity', MAX(sm.tbs)) AS diagnostic_info FROM StabilityMetrics sm LEFT JOIN container c ON sm.containregistry = c.containregistry GROUP BY sm.containregistry, c.coolremainpct ) SELECT containregistry, ROUND(tss, 3) AS max_temp_stability_score, ROUND(cri, 3) AS container_risk_index, risk_rank, sensor_data_status, diagnostic_info FROM RiskIndex WHERE cri > 0.6 OR tss < 0.4 ORDER BY risk_rank LIMIT 10;
