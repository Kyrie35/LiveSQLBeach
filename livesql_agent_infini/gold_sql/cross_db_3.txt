WITH FlowMetrics AS ( SELECT df.RecordRegistry, df.BwidthPct * df.DataSizeMB / df.DurMin AS bsi, COALESCE(df.SuccessPct / NULLIF(df.ErrTally + 1, 0), 0) AS dte FROM DataFlow AS df WHERE df.BwidthPct * df.DataSizeMB / df.DurMin > 50 AND COALESCE(df.SuccessPct / NULLIF(df.ErrTally + 1, 0), 0) < 1.0 ) SELECT RecordRegistry, ROUND(CAST(bsi AS REAL), 2) AS bandwidth_saturation_index, ROUND(CAST(dte AS REAL), 2) AS data_transfer_efficiency FROM FlowMetrics ORDER BY bsi DESC NULLS FIRST LIMIT 10;
