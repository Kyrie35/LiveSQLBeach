WITH AuditSeverity AS ( SELECT ac.AuditTrace, COALESCE(CAST(ac.CritFindNum AS REAL) / NULLIF(ac.FindTally + 1, 0), 0) AS afs, ac.RemedDue FROM AuditAndCompliance AS ac WHERE NOT ac.RemedDue IS NULL ), MonthlyTrends AS ( SELECT STRFTIME('%Y-%m', RemedDue) AS audit_month, AVG(afs) AS avg_severity, DENSE_RANK() OVER (ORDER BY AVG(afs) DESC NULLS FIRST) AS severity_rank FROM AuditSeverity WHERE afs > 0.5 GROUP BY STRFTIME('%Y-%m', RemedDue) ) SELECT audit_month, ROUND(CAST(avg_severity AS REAL), 2) AS average_afs, severity_rank FROM MonthlyTrends WHERE severity_rank <= 3 ORDER BY audit_month NULLS LAST;
