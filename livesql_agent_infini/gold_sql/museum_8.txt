SELECT a.ArtRegistry, a.ArtName, a.MatType, sd.LightSensitivity, CASE WHEN ( SELECT MAX(l.LightLux) FROM LightAndRadiationReadings AS l JOIN EnvironmentalReadingsCore AS e ON l.EnvReadRef = e.EnvReadRegistry JOIN UsageRecords AS u ON e.ShowcaseRef = u.ShowcaseRefUsed WHERE u.ArtRefUsed = a.ArtRegistry ) > CASE sd.LightSensitivity WHEN 'High' THEN 50 WHEN 'Medium' THEN 200 ELSE 300 END THEN 'Over Exposure' ELSE 'Within Limits' END AS light_exposure_status FROM ArtifactsCore AS a JOIN SensitivityData AS sd ON a.ArtRegistry = sd.ArtRef WHERE sd.LightSensitivity IN ('High', 'Medium');
