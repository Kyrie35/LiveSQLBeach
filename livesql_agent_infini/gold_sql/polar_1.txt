WITH equipment_efficiency AS ( SELECT equipmentcode, equipmenttype, CAST(( ( performanceindex + reliabilityindex ) / 2 * ( 1 - environmentalimpactindex / 10 ) ) AS REAL(10, 2)) AS eer, safetyindex FROM equipment ) SELECT equipmenttype, equipmentcode, eer, safetyindex, RANK() OVER (PARTITION BY equipmenttype ORDER BY eer DESC NULLS FIRST) AS efficiency_rank, RANK() OVER (PARTITION BY equipmenttype ORDER BY safetyindex DESC NULLS FIRST) AS safety_rank, ABS( RANK() OVER (PARTITION BY equipmenttype ORDER BY eer DESC NULLS FIRST) - RANK() OVER (PARTITION BY equipmenttype ORDER BY safetyindex DESC NULLS FIRST) ) AS rank_difference FROM equipment_efficiency ORDER BY equipmenttype NULLS LAST, eer DESC NULLS FIRST;
