WITH RiskVolume AS ( SELECT df.RecordRegistry, ( rm.RiskAssess / rm.CtrlEff ) * CASE WHEN df.OrigNation <> df.DestNation THEN 2 ELSE 1 END AS cbrf, ( ( rm.RiskAssess / rm.CtrlEff ) * CASE WHEN df.OrigNation <> df.DestNation THEN 2 ELSE 1 END ) * dp.VolGB AS cdvr FROM DataFlow AS df INNER JOIN RiskManagement AS rm ON df.RecordRegistry = rm.FlowLink INNER JOIN DataProfile AS dp ON df.RecordRegistry = dp.FlowSign WHERE df.OrigNation <> df.DestNation ) SELECT RecordRegistry, ROUND(CAST(cbrf AS REAL), 2) AS cross_border_risk_factor, ROUND(CAST(cdvr AS REAL), 2) AS cross_border_data_volume_risk, ROW_NUMBER() OVER (ORDER BY cdvr DESC NULLS FIRST) AS risk_rank FROM RiskVolume WHERE cdvr > 1000 ORDER BY cdvr DESC NULLS FIRST LIMIT 5;
