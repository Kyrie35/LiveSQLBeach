WITH engagement_data AS ( SELECT nm.netkey, nm.netsessref, CAST(( nm.network_engagement_metrics -> '$.engagement_metrics' ->> '$.engauth' ) AS REAL) AS engauth, ( nm.network_engagement_metrics -> '$.interaction_metrics' ->> '$.tempinteractpat' ) AS tempinteractpat FROM networkmetrics AS nm ) SELECT a.acctident, ed.engauth, ed.tempinteractpat, CASE WHEN ed.engauth < 0.3 AND ed.tempinteractpat = 'Automated' THEN 'Manipulator' ELSE 'Normal' END AS manipulator_status FROM engagement_data AS ed LEFT JOIN sessionbehavior AS sb ON ed.netsessref = sb.sessref LEFT JOIN profile AS p ON sb.sessprofref = p.profkey LEFT JOIN account AS a ON p.profaccref = a.accindex WHERE ed.engauth < 0.3 AND ed.tempinteractpat = 'Automated' LIMIT 100;
