WITH risky_chains AS ( SELECT r.riskregistry, r.txchainlen, r.linkedtxcount, r.wallage, r.wallturnrt, ( r.txchainlen * 10 ) + ( r.linkedtxcount * 5 ) + ( r.fraudprob * 100 ) - ( r.profilecomplete * 0.5 ) - ( r.idverifyscore * 0.5 ) AS transaction_chain_risk FROM riskanalysis AS r ) SELECT rc.riskregistry, rc.transaction_chain_risk, t.txregistry, t.payamtusd FROM risky_chains AS rc JOIN riskanalysis AS ra ON rc.riskregistry = ra.riskregistry JOIN transactions AS t ON ra.txref = t.txregistry WHERE rc.transaction_chain_risk > 150 AND rc.wallage < 30 AND rc.wallturnrt > 5 AND rc.linkedtxcount >= 3 ORDER BY rc.transaction_chain_risk DESC NULLS FIRST;
