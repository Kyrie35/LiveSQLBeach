WITH artifact_rankings AS ( SELECT a.ArtRegistry, a.ArtName, a.ConserveStatus, sec.InsValueUSD, r.HistSignRating, r.CultScore, NTILE(10) OVER (ORDER BY r.HistSignRating DESC NULLS FIRST) AS hist_percentile, NTILE(10) OVER (ORDER BY r.CultScore DESC NULLS FIRST) AS cult_percentile FROM ArtifactsCore AS a JOIN ArtifactRatings AS r ON a.ArtRegistry = r.ArtRef JOIN ArtifactSecurityAccess AS sec ON a.ArtRegistry = sec.ArtRef ) SELECT ArtRegistry, ArtName, InsValueUSD, ConserveStatus, CASE WHEN InsValueUSD > 1000000 THEN 'Monetary High-Value' WHEN hist_percentile = 1 AND cult_percentile = 1 THEN 'Cultural/Historical High-Value' ELSE 'Other' END AS value_category FROM artifact_rankings WHERE ConserveStatus = 'Critical' AND ( InsValueUSD > 1000000 OR ( hist_percentile = 1 AND cult_percentile = 1 ) );
