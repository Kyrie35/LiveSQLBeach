WITH TransactionSentiment AS ( SELECT tr.transreg, tr.trdref, tr.transtime, sf.* FROM transactionrecord AS tr INNER JOIN sentimentandfundamentals AS sf ON tr.transreg = sf.transref ), InsiderFlaggedTransactions AS ( SELECT transreg, trdref, transtime, infoleaksc, optvolrt, newsscore, socscore FROM TransactionSentiment WHERE infoleaksc > 50.0 AND NOT corpeventprx IS NULL AND eventannotm IN ('Pre-market', 'Intraday') ), CalculatedRisks AS ( SELECT ift.*, ABS(ift.newsscore - ift.socscore) AS sdf, ift.optvolrt * ( 1 + ABS(ift.newsscore - ift.socscore) ) AS swov FROM InsiderFlaggedTransactions AS ift ), SDLR_Calculation AS ( SELECT cr.*, ROUND(cr.swov * cr.infoleaksc, 2) AS sdlr FROM CalculatedRisks AS cr ) SELECT transreg, trdref, transtime, infoleaksc, sdlr FROM SDLR_Calculation WHERE sdlr > 1000 LIMIT 100;
