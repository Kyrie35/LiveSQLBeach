WITH AuditPressure AS ( SELECT df.RecordRegistry AS flow_id, ((CAST(ac.CritFindNum AS REAL) / (ac.FindTally + 1)) * (ac.AccReqNum + ac.DelReqNum + ac.RectReqNum + ac.PortReqNum) * (CAST(ac.CritFindNum AS REAL) / (ac.FindTally + 1))) AS acp, JULIANDAY('2025-04-01') - JULIANDAY(ac.RemedDue) AS days_overdue FROM DataFlow df JOIN RiskManagement rm ON df.RecordRegistry = rm.FlowLink JOIN DataProfile dp ON dp.FlowSign = df.RecordRegistry JOIN AuditAndCompliance ac ON ac.ProfJoin = dp.ProfileTrace WHERE df.OrigNation != df.DestNation AND ac.RemedDue IS NOT NULL ) SELECT ap.flow_id, ROUND(ap.acp, 2) AS audit_compliance_pressure, ap.days_overdue FROM AuditPressure ap WHERE ap.acp > 5 AND ap.days_overdue < -5 ORDER BY ap.days_overdue DESC, ap.acp DESC LIMIT 10;
