WITH PatientRisk AS ( SELECT ab.patownerref AS patkey, asr.asrkey, asr.suicrisk, CAST(( asr.mental_health_scores -> '$.depression' ->> '$.phq9_score' ) AS INTEGER) AS phq9_score, CAST(( asr.mental_health_scores -> '$.anxiety' ->> '$.gad7_score' ) AS INTEGER) AS gad7_score, TRUE AS is_high_risk FROM assessmentbasics AS ab JOIN assessmentsymptomsandrisk AS asr ON ab.abkey = asr.asrkey WHERE asr.suicrisk IN ('High', 'Severe') OR CAST(( asr.mental_health_scores -> '$.depression' ->> '$.phq9_score' ) AS INTEGER) > 15 OR CAST(( asr.mental_health_scores -> '$.anxiety' ->> '$.gad7_score' ) AS INTEGER) > 15 ), FacilityStress AS ( SELECT fackey, envstress, lifeimpact FROM facilities WHERE envstress = 'Severe' OR lifeimpact = 'Severe' ), PatientEngagement AS ( SELECT tb.encref, e.patref AS patkey, CASE ( tb.therapy_details ->> '$.engagement' ) WHEN 'High' THEN 3 WHEN 'Medium' THEN 2 WHEN 'Low' THEN 1 WHEN 'Non-compliant' THEN 0 ELSE 0 END AS engagement_score FROM treatmentbasics AS tb JOIN encounters AS e ON tb.encref = e.enckey WHERE NOT tb.therapy_details IS NULL ), AvgPatientEngagement AS ( SELECT patkey, ROUND(AVG(engagement_score)) AS avg_engagement_score FROM PatientEngagement GROUP BY patkey ), RecentAssessment AS ( SELECT pr.patkey, pr.asrkey, pr.is_high_risk, e.facid, e.timemark, ROW_NUMBER() OVER (PARTITION BY pr.patkey ORDER BY e.timemark DESC NULLS FIRST) AS rn FROM PatientRisk AS pr JOIN encounters AS e ON pr.asrkey = e.abref ) SELECT ra.patkey, ra.asrkey, ra.timemark AS last_assessment_date, ape.avg_engagement_score AS patient_tes_avg, fs.envstress, fs.lifeimpact FROM RecentAssessment AS ra JOIN AvgPatientEngagement AS ape ON ra.patkey = ape.patkey LEFT JOIN FacilityStress AS fs ON ra.facid = fs.fackey WHERE ra.rn = 1 AND ape.avg_engagement_score < 2 AND NOT fs.fackey IS NULL ORDER BY ra.timemark DESC NULLS FIRST LIMIT 50;
