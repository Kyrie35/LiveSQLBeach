WITH inverter_loss AS ( SELECT p.solmodref, ROUND( p.measpoww * ( 1 - CAST(( i.power_metrics ->> '$.inverteffpct' ) AS REAL) / 100 ), 2 ) AS IEL FROM performance AS p INNER JOIN panel AS pa ON p.solmodref = pa.panemark INNER JOIN inverter AS i ON i.siteref = pa.hubregistry ) SELECT p.solmodref AS panel_id, pa.powratew AS rated_power, p.measpoww AS measured_power, ROUND( pa.powratew * ( CAST(( p.efficiency_profile -> '$.degradation' ->> '$.cumdegpct' ) AS REAL) / 100 ), 2 ) AS degradation_loss, ROUND(p.measpoww * ( env.soillosspct / 100.0 ), 2) AS soiling_loss, inv.IEL AS inverter_loss, ROUND( ( pa.powratew * ( CAST(( p.efficiency_profile -> '$.degradation' ->> '$.cumdegpct' ) AS REAL) / 100 ) ) + ( p.measpoww * ( env.soillosspct / 100.0 ) ) + inv.IEL, 2 ) AS TSL FROM performance AS p INNER JOIN panel AS pa ON p.solmodref = pa.panemark INNER JOIN environment AS env ON env.arearegistry = pa.hubregistry INNER JOIN inverter_loss AS inv ON inv.solmodref = p.solmodref;
