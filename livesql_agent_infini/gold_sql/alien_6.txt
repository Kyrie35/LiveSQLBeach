WITH priority_calc AS ( SELECT s.SignalRegistry, ( p.TechSigProb * 4 + p.BioSigProb / 100 + p.SigUnique * 2 + p.AnomScore / 2 ) * ( 1 - p.FalsePosProb ) AS RPI, ( 1 - p.FalsePosProb ) * d.DecodeConf * ( CASE WHEN s.SnrRatio - 0.1 * ABS(s.NoiseFloorDbm) > 0 THEN ( s.SnrRatio - 0.1 * ABS(s.NoiseFloorDbm) ) / 10 + 0.5 ELSE 0.1 END ) AS CCS_approx, o.ObservStation FROM Signals AS s JOIN SignalProbabilities AS p ON s.SignalRegistry = p.SignalRef JOIN SignalDecoding AS d ON s.SignalRegistry = d.SignalRef JOIN Telescopes AS t ON s.TelescRef = t.TelescRegistry JOIN Observatories AS o ON t.ObservStation = o.ObservStation ) SELECT ObservStation, COUNT(*) AS signal_count, AVG(RPI) AS avg_rpi, AVG(CCS_approx) AS avg_ccs, COUNT(*) FILTER(WHERE RPI > 3) AS high_priority_signals, COUNT(*) FILTER(WHERE CCS_approx > 0.8) AS high_confidence_signals, COUNT(*) FILTER(WHERE RPI > 3 AND CCS_approx > 0.8) AS high_priority_high_confidence FROM priority_calc GROUP BY ObservStation ORDER BY high_priority_high_confidence DESC NULLS FIRST;
