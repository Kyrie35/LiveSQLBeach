WITH panel_base_data AS ( SELECT p.panemark, p.hubregistry, p.paneeffpct AS initial_efficiency, plant.initdate AS installation_date, plant.growalias AS plant_name, CAST(strftime('%Y', 'now') AS INTEGER) - CAST(strftime('%Y', plant.initdate) AS INTEGER) AS years_in_service FROM panel p JOIN plant ON p.hubregistry = plant.growregistry WHERE plant.initdate IS NOT NULL ), efficiency_data AS ( SELECT pbd.panemark, pbd.initial_efficiency, pbd.installation_date, pbd.plant_name, pbd.years_in_service, perf.perfmoment, REPLACE(REPLACE(JSON_EXTRACT(perf.efficiency_profile, '$.current_efficiency.curreffpct'), '[', ''), ']', '') AS current_efficiency, REPLACE(REPLACE(JSON_EXTRACT(perf.efficiency_profile, '$.degradation.anndegrate'), '[', ''), ']', '') AS annual_deg_rate, CAST(strftime('%Y', perf.perfmoment) AS INTEGER) - CAST(strftime('%Y', pbd.installation_date) AS INTEGER) AS measurement_age_years, ROW_NUMBER() OVER (PARTITION BY perf.solmodref ORDER BY perf.perfmoment DESC) AS rn FROM panel_base_data pbd JOIN performance perf ON pbd.panemark = perf.solmodref WHERE perf.efficiency_profile IS NOT NULL ), degradation_analysis AS ( SELECT ed.panemark, ed.initial_efficiency, ed.installation_date, ed.plant_name, ed.years_in_service, ed.current_efficiency, ed.annual_deg_rate, CASE WHEN ed.measurement_age_years > 0 THEN (ed.initial_efficiency - ed.current_efficiency) / ed.measurement_age_years ELSE NULL END AS pelr FROM efficiency_data ed WHERE ed.rn = 1 ), normalized_degradation AS ( SELECT da.panemark, da.initial_efficiency, da.installation_date, da.plant_name, da.years_in_service, da.current_efficiency, da.annual_deg_rate, da.pelr, CASE WHEN da.annual_deg_rate > 0 THEN da.pelr / da.annual_deg_rate ELSE NULL END AS ndi FROM degradation_analysis da ) SELECT nd.plant_name, nd.panemark AS panel_id, strftime('%Y-%m-%d', nd.installation_date) AS installation_date, nd.years_in_service, ROUND(nd.initial_efficiency, 2) AS initial_efficiency_pct, ROUND(nd.current_efficiency, 2) AS current_efficiency_pct, ROUND(nd.annual_deg_rate, 2) AS annual_degradation_rate_pct, ROUND(nd.pelr * 100, 2) AS panel_efficiency_loss_rate_pct, ROUND(nd.ndi, 2) AS normalized_degradation_index, CASE WHEN nd.ndi > 1.5 THEN 'High Degradation' WHEN nd.ndi > 1.0 THEN 'Moderate Degradation' ELSE 'Normal Degradation' END AS degradation_level FROM normalized_degradation nd;
