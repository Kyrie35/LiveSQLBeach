WITH bbi_calc AS ( SELECT m.macntref AS accindex, ( 0.4 * CAST(( sd.detection_score_profile -> '$.behavior_scores' ->> '$.botlikscore' ) AS REAL) + 0.3 * CAST(( sd.detection_score_profile -> '$.behavior_scores' ->> '$.autobehavscore' ) AS REAL) + 0.3 * ( 1 - ma.convnatval ) ) AS bbi_score FROM moderationaction AS m FULL JOIN securitydetection AS sd ON m.masedetref = sd.secdetkey JOIN messaginganalysis AS ma ON m.macntref = ma.msgcntref ), cluster_stats AS ( SELECT m.clustsize, ROUND(AVG(COALESCE(bbi_score, 0)), 2) AS avg_bbi, COUNT(m.modactkey) AS account_count FROM moderationaction AS m FULL JOIN bbi_calc AS bc ON m.macntref = bc.accindex GROUP BY m.clustsize HAVING m.clustsize > 10 ) SELECT clustsize, avg_bbi, account_count FROM cluster_stats WHERE avg_bbi > 0.7 ORDER BY avg_bbi DESC NULLS FIRST LIMIT 100;
