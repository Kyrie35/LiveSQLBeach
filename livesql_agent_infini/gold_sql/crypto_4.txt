SELECT r.riskandmarginpivot, o.recordvault, ab.accountbalancesnode, CAST(( r.risk_margin_profile -> '$.margin_thresholds' ->> '$.inithold' ) AS REAL) AS inithold, CAST(ab.margsum AS REAL), ( CAST(( r.risk_margin_profile -> '$.margin_thresholds' ->> '$.inithold' ) AS REAL) / CAST(ab.margsum AS REAL) * 100 ) AS margin_utilization FROM riskandmargin AS r JOIN orders AS o ON r.ordervault = o.recordvault JOIN accountbalances AS ab ON o.userlink = ab.usertag;
