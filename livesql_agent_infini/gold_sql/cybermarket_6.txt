WITH market_escrow_rates AS ( SELECT mktregistry, esccomprate FROM markets ) SELECT t.txregistry, t.paymethod, t.payamtusd, t.escrowused, t.escrowhrs, t.routecomplexity, ( t.payamtusd / 1000 * t.txfinishhrs / 24 * ( 1 + CAST(t.escrowhrs AS REAL) / 100 ) * ( 1 - mer.esccomprate / 100 ) ) AS transaction_anomaly_score FROM transactions AS t JOIN market_escrow_rates AS mer ON t.mktref = mer.mktregistry WHERE ( t.payamtusd / 1000 * t.txfinishhrs / 24 * ( 1 + CAST(t.escrowhrs AS REAL) / 100 ) * ( 1 - mer.esccomprate / 100 ) ) > 75 AND t.paymethod = 'Crypto_B' AND ( t.escrowused = 'No' OR t.escrowhrs < 24 ) AND t.routecomplexity = 'Complex' ORDER BY transaction_anomaly_score DESC NULLS FIRST;
