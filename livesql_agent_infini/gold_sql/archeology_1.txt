SELECT s.zoneref AS site_code, si.zonelabel AS site_name, ROUND( AVG( POWER( 10.0 / ( LOG(10, spc.scanresolmm * 1000.0) / LOG(10, spc.pointdense) ) * 5.0, 1.5 ) * ( spc.coverpct / 100.0 ) * POWER(1.0 - spc.noisedb / 30.0, 2) ), 2 ) AS avg_sqs FROM scans AS s JOIN scanpointcloud AS spc ON s.arcref = spc.arcref JOIN sites AS si ON s.zoneref = si.zoneregistry GROUP BY s.zoneref, si.zonelabel ORDER BY avg_sqs DESC NULLS FIRST;
