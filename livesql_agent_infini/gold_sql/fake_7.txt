WITH dormant_risk AS ( SELECT a.accindex, m.reputscore, m.abuserepnum, MAX(m.lastrevdate) AS last_review_date FROM account AS a JOIN profile AS p ON p.profaccref = a.accindex JOIN sessionbehavior AS sb ON sb.sessprofref = p.profkey JOIN contentbehavior AS cb ON cb.cntsessref = sb.sessref JOIN moderationaction AS m ON m.macntref = cb.cntref WHERE a.acctstatus = 'Dormant' GROUP BY a.accindex, m.reputscore, m.abuserepnum ), risky_dormants AS ( SELECT accindex, reputscore, abuserepnum, last_review_date, NTILE(4) OVER (ORDER BY abuserepnum DESC NULLS FIRST) AS risk_quartile FROM dormant_risk WHERE reputscore < 30 ) SELECT accindex, reputscore, abuserepnum, last_review_date FROM risky_dormants WHERE risk_quartile = 1 ORDER BY abuserepnum DESC NULLS FIRST LIMIT 10;
