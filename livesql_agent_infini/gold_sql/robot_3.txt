WITH SafetyIncidents AS ( SELECT ps.effectivenessrobot, SUM( COALESCE(CAST(( ps.safety_metrics ->> '$.overloads' ) AS INTEGER), 0) + COALESCE(CAST(( ps.safety_metrics ->> '$.collisions' ) AS INTEGER), 0) + COALESCE(CAST(( ps.safety_metrics ->> '$.emergency_stops' ) AS INTEGER), 0) + COALESCE(CAST(( ps.safety_metrics ->> '$.speed_violations' ) AS INTEGER), 0) ) AS sis FROM performance_and_safety AS ps GROUP BY ps.effectivenessrobot ) SELECT rd.bottypeval AS robot_type, AVG(si.sis) AS avg_sis, MAX(si.sis) AS max_sis FROM SafetyIncidents AS si INNER JOIN robot_details AS rd ON si.effectivenessrobot = rd.botdetreg WHERE si.sis > 20 GROUP BY rd.bottypeval ORDER BY avg_sis DESC NULLS FIRST LIMIT 100;
