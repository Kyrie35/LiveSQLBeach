WITH FacilityResourceScores AS ( SELECT fackey, CASE ( support_and_resources ->> '$.community_resources' ) WHEN 'Comprehensive' THEN 3 WHEN 'Adequate' THEN 2 WHEN 'Limited' THEN 1 ELSE 0 END AS resource_score FROM facilities WHERE NOT support_and_resources ->> '$.community_resources' IS NULL ), ResourceSupportedFacilities AS ( SELECT fackey FROM FacilityResourceScores WHERE resource_score >= 2 ), PatientMetrics AS ( SELECT e.patref AS patkey, e.clinid, e.facid, COALESCE(tb.crisisint, 0) AS crisisint, COALESCE(e.missappt, 0) AS missappt, e.enckey FROM encounters AS e LEFT JOIN treatmentbasics AS tb ON e.enckey = tb.encref WHERE NOT e.clinid IS NULL AND e.facid IN ( SELECT fackey FROM ResourceSupportedFacilities ) ), ClinicianAvgMetrics AS ( SELECT clinid, facid, SUM(crisisint) / CAST(NULLIF(COUNT(DISTINCT patkey), 0) AS REAL) AS avg_cif, SUM(missappt) / CAST(NULLIF(COUNT(DISTINCT patkey), 0) AS REAL) AS avg_mar FROM PatientMetrics GROUP BY clinid, facid ), ClinicianPSM AS ( SELECT clinid, facid, avg_cif, avg_mar, ROUND(1.0 / ( 1.0 + avg_cif + avg_mar ), 2) AS psm FROM ClinicianAvgMetrics ) SELECT clinid, facid, psm, RANK() OVER (PARTITION BY facid ORDER BY psm DESC NULLS FIRST, clinid NULLS LAST) AS rank_within_facility FROM ClinicianPSM ORDER BY facid NULLS LAST, rank_within_facility NULLS LAST LIMIT 100;
