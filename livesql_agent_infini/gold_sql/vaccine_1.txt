WITH SensorMetrics AS ( SELECT c.containregistry, (1 - COALESCE(CAST(s.tempdevcount AS REAL) / 100.0, 0)) * (1 - COALESCE(CAST(s.critevents AS REAL) / 10.0, 0)) * COALESCE(s.tempstabidx, 1.0) AS tss, COALESCE(ABS(s.tempnowc - s.storetempc) / NULLIF(s.temptolc, 0) * COALESCE(s.tempdevcount, 0), 0.0) AS tbs, s.alerttime, (1 - ((1 - COALESCE(CAST(s.tempdevcount AS REAL) / 100.0, 0)) * (1 - COALESCE(CAST(s.critevents AS REAL) / 10.0, 0)) * COALESCE(s.tempstabidx, 1.0))) * (1 - COALESCE(c.coolremainpct, 100.0) / 100.0) AS cri FROM container c LEFT JOIN sensordata s ON c.containregistry = s.containlink AND s.alerttime >= date((SELECT MAX(alerttime) FROM sensordata), '-1 year') ), StabilityMetrics AS ( SELECT sm.containregistry, COALESCE(AVG(sm.tss), 1.0) AS avg_tss, COALESCE(MAX(sm.tbs), 0.0) AS max_tbs, COALESCE(MAX(CASE WHEN sm.alerttime = (SELECT MAX(sm2.alerttime) FROM SensorMetrics sm2 WHERE sm2.containregistry = sm.containregistry) THEN sm.tss END), 1.0) AS current_tss, COALESCE(MAX(CASE WHEN sm.alerttime = (SELECT MAX(sm2.alerttime) FROM SensorMetrics sm2 WHERE sm2.containregistry = sm.containregistry) THEN sm.cri END), 0.0) AS cri FROM SensorMetrics sm GROUP BY sm.containregistry ) SELECT containregistry, ROUND(avg_tss, 2) AS avg_temp_stability_score, ROUND(max_tbs, 2) AS max_temp_breach_severity, CASE WHEN cri > 0.6 OR current_tss < 0.4 THEN 'Critical' WHEN avg_tss < 0.4 OR max_tbs > 1.5 THEN 'Unstable' WHEN avg_tss >= 0.7 AND max_tbs <= 1.0 THEN 'Stable' ELSE 'Moderate' END AS container_health_status FROM StabilityMetrics LIMIT 100;
