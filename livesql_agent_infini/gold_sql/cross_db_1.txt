WITH HighRiskFlows AS ( SELECT df.RecordRegistry, COALESCE(rm.RiskAssess / NULLIF(rm.CtrlEff, 0), 0) AS res, COALESCE(dp.VolGB, 0) * CASE WHEN dp.DataSense = 'High' THEN 3 WHEN dp.DataSense = 'Medium' THEN 2 ELSE 1 END AS dsi FROM DataFlow AS df LEFT JOIN RiskManagement AS rm ON df.RecordRegistry = rm.FlowLink LEFT JOIN DataProfile AS dp ON df.RecordRegistry = dp.FlowSign WHERE COALESCE(rm.RiskAssess / NULLIF(rm.CtrlEff, 0), 0) > 0.7 AND COALESCE(dp.VolGB, 0) * CASE WHEN dp.DataSense = 'High' THEN 3 WHEN dp.DataSense = 'Medium' THEN 2 ELSE 1 END > 100 ), RankedFlows AS ( SELECT RecordRegistry, res, dsi, RANK() OVER (ORDER BY dsi DESC NULLS FIRST) AS volume_rank FROM HighRiskFlows ) SELECT RecordRegistry, ROUND(CAST(res AS REAL), 2) AS risk_exposure_score, ROUND(CAST(dsi AS REAL), 2) AS data_sensitivity_index FROM RankedFlows WHERE volume_rank <= 5 ORDER BY volume_rank NULLS LAST LIMIT 5;
