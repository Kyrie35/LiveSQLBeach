WITH quality_assessment AS ( SELECT o.ObservStation, t.TelescRegistry, o.AtmosTransparency, o.HumidityRate, o.WindSpeedMs, o.LunarDistDeg, t.EquipStatus, t.CalibrStatus, t.PointAccArc, (o.AtmosTransparency * (1 - o.HumidityRate/100) * (1 - 0.02 * o.WindSpeedMs)) AS AOI, ((1 - o.LunarDistDeg/180) * (1 - o.AtmosTransparency)) AS LIF, CASE WHEN t.EquipStatus != 'Degraded' THEN t.EquipStatus ELSE '' END || ',' || CASE WHEN t.CalibrStatus != 'Overdue' THEN t.CalibrStatus ELSE '' END || ',' || CASE WHEN t.CoolSysStatus != 'Warning' THEN t.CoolSysStatus ELSE '' END AS equipment_issues_concat FROM Observatories o INNER JOIN Telescopes t ON o.ObservStation = t.ObservStation ), equipment_issues_split AS ( SELECT ObservStation, TelescRegistry, AOI, LIF, PointAccArc, CASE WHEN LENGTH(equipment_issues_concat) - LENGTH(REPLACE(equipment_issues_concat, ',', '')) > 0 THEN REPLACE(equipment_issues_concat, ',NULL', '') ELSE equipment_issues_concat END AS equipment_issues_cleaned FROM quality_assessment ) SELECT qa.ObservStation, qa.TelescRegistry, (qa.AOI * (1 - qa.LIF) * (CASE WHEN qa.PointAccArc < 2 THEN 1 ELSE 2/qa.PointAccArc END)) AS OQF, CASE WHEN qa.equipment_issues_cleaned = ',,' THEN '' ELSE REPLACE(qa.equipment_issues_cleaned, ',,', ',') END AS identified_issues, CASE WHEN qa.equipment_issues_cleaned = ',,' THEN 0 ELSE LENGTH(qa.equipment_issues_cleaned) - LENGTH(REPLACE(qa.equipment_issues_cleaned, ',', '')) END AS issue_count, CASE WHEN qa.AOI > 0.8 AND quality_assessment.EquipStatus = 'Operational' AND quality_assessment.CalibrStatus = 'Current' THEN 'High' WHEN qa.AOI BETWEEN 0.5 AND 0.8 THEN 'Medium' ELSE 'Low' END AS ocl_classification FROM equipment_issues_split qa INNER JOIN quality_assessment ON qa.ObservStation = quality_assessment.ObservStation ORDER BY OQF DESC;
