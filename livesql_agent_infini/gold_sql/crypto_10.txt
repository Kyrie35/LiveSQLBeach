SELECT o.orderspivot, o.dealedge, o.dealcount, md.quote_depth_snapshot -> '$.depth' AS depth FROM orders AS o JOIN marketdata AS md ON md.marketdatanode = ( SELECT mdlink FROM marketstats WHERE marketstatsmark = ( SELECT MAX(marketstatsmark) FROM marketstats ) ) WHERE ( ( o.dealedge = 'Sell' AND o.dealcount > 0.1 * CAST(( md.quote_depth_snapshot -> '$.depth' ->> '$.bidunits' ) AS REAL) ) OR ( o.dealedge = 'Buy' AND o.dealcount > 0.1 * CAST(( md.quote_depth_snapshot -> '$.depth' ->> '$.askunits' ) AS REAL) ) );
