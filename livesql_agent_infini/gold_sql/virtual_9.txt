SELECT f.userregistry, f.nicklabel, COUNT(DISTINCT i.interactidolpivot) AS idols_supported, GROUP_CONCAT(DISTINCT vi.nametag) AS idol_names, ROUND(AVG(e.engrate), 2) AS avg_engagement_rate FROM fans f JOIN interactions i ON i.interactfanpivot = f.userregistry JOIN virtualidols vi ON vi.entityreg = i.interactidolpivot JOIN membershipandspending ms ON ms.memberfanpivot = f.userregistry JOIN engagement e ON e.engagememberpivot = ms.memberreg WHERE e.engrate > 0.4 GROUP BY f.userregistry, f.nicklabel HAVING COUNT(DISTINCT i.interactidolpivot) >= 2 ORDER BY idols_supported DESC, avg_engagement_rate DESC;
