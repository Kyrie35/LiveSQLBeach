SELECT sp.flowsoft AS processing_software, sp.flowstage AS processing_stage, ROUND(CAST(AVG(sp.flowhrs) AS REAL), 1) AS avg_processing_hours, ROUND(CAST(AVG(sp.proccpu) AS REAL), 0) AS avg_cpu_usage_pct, ROUND(CAST(AVG(sp.procgpu) AS REAL), 0) AS avg_gpu_usage_pct, ROUND(CAST(AVG(s.gbsize) AS REAL), 1) AS avg_data_size_gb, ROUND( CAST(AVG( s.gbsize * LOG(10, COALESCE(spc.totalpts, 1000000)) / ( sp.flowhrs * ( sp.proccpu + sp.procgpu ) / 200 ) ) AS REAL), 1 ) AS avg_per, CASE WHEN AVG( s.gbsize * LOG(10, COALESCE(spc.totalpts, 1000000)) / ( sp.flowhrs * ( sp.proccpu + sp.procgpu ) / 200 ) ) < 0.5 THEN 'Bottleneck Detected' ELSE 'Efficient' END AS efficiency_status, COUNT(*) AS workflow_count FROM scanprocessing AS sp JOIN scans AS s ON sp.zoneref = s.zoneref LEFT JOIN scanpointcloud AS spc ON s.arcref = spc.arcref WHERE sp.flowhrs > 0 AND ( sp.proccpu + sp.procgpu ) > 0 GROUP BY sp.flowsoft, sp.flowstage ORDER BY CASE WHEN AVG( s.gbsize * LOG(10, COALESCE(spc.totalpts, 1000000)) / ( sp.flowhrs * ( sp.proccpu + sp.procgpu ) / 200 ) ) < 0.5 THEN 1 ELSE 2 END NULLS LAST, avg_per NULLS LAST;
