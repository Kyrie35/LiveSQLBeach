WITH fan_memberships AS ( SELECT f.userregistry, f.nicklabel, ms.memberreg, ms.membdays FROM fans AS f JOIN membershipandspending AS ms ON ms.memberfanpivot = f.userregistry ), fan_engagement AS ( SELECT fm.userregistry, fm.nicklabel, fm.membdays, e.engagereg, e.engrate, e.socintscore, e.actdayswk, e.avgsesscount, e.actfreq, e.engrate * 0.4 + e.socintscore / 100 * 0.3 + e.actdayswk / 7.0 * 0.2 + e.avgsesscount / 10.0 * 0.1 AS fei_approx FROM fan_memberships AS fm JOIN engagement AS e ON e.engagememberpivot = fm.memberreg ), community_data AS ( SELECT fe.userregistry, fe.nicklabel, fe.membdays, fe.actfreq, fe.fei_approx, sc.collabcount, sc.community_engagement, CAST(( sc.community_engagement -> '$.group_involvement' ->> '$.grprole' ) AS TEXT) AS group_role, CAST(( sc.community_engagement -> '$.content_creation' ->> '$.contqualrate' ) AS REAL) AS contqualrate, CAST(( sc.community_engagement -> '$.content_creation' ->> '$.ugcval' ) AS REAL) AS ugcval, CAST(( sc.community_engagement -> '$.network' ->> '$.follcount' ) AS REAL) AS follcount, ri.viralcont FROM fan_engagement AS fe JOIN socialcommunity AS sc ON sc.socialengagepivot = fe.engagereg JOIN retentionandinfluence AS ri ON ri.retainengagepivot = fe.engagereg ), cci_components AS ( SELECT cd.userregistry, cd.nicklabel, cd.contqualrate * ( cd.ugcval / 10.0 ) * ( 1 + cd.follcount / 100 * 0.5 ) AS ccis_approx, ( cd.follcount / 100 ) * ( fe.engrate * 2 ) * ( COALESCE(cd.viralcont, 0) + 1 ) * 0.5 AS sim_approx, cd.collabcount, cd.fei_approx, cd.actfreq, cd.membdays, cd.group_role FROM community_data AS cd JOIN fan_engagement AS fe ON fe.userregistry = cd.userregistry ) SELECT nicklabel, ROUND( CAST(( ccis_approx * 0.4 + sim_approx * 0.3 + collabcount * 0.1 + fei_approx * 0.2 ) AS REAL), 2 ) AS cci, group_role, actfreq, membdays FROM cci_components WHERE ( ccis_approx * 0.4 + sim_approx * 0.3 + collabcount * 0.1 + fei_approx * 0.2 ) > 7 AND actfreq = 'Daily' AND membdays > 180 AND ( group_role = 'Moderator' OR group_role = 'Leader' ) ORDER BY cci DESC NULLS FIRST;
