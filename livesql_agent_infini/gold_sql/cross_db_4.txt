SELECT JSON_OBJECT( 'ProfileTrace', dp.ProfileTrace, 'DataSensitivityIndex', ROUND((dp.VolGB * CASE WHEN dp.DataSense = 'High' THEN 3 WHEN dp.DataSense = 'Medium' THEN 2 ELSE 1 END), 2), 'SecurityRobustnessScore', CASE WHEN sp.EncState = 'Full' AND sp.AclState = 'Strong' THEN 3 WHEN sp.EncState = 'Full' OR sp.AclState = 'Strong' THEN 2 ELSE 1 END ) AS exposure_details FROM DataProfile dp LEFT JOIN SecurityProfile sp ON dp.ProfileTrace = sp.ProfileKey WHERE dp.VolGB * CASE WHEN dp.DataSense = 'High' THEN 3 WHEN dp.DataSense = 'Medium' THEN 2 ELSE 1 END > 100 AND CASE WHEN sp.EncState = 'Full' AND sp.AclState = 'Strong' THEN 3 WHEN sp.EncState = 'Full' OR sp.AclState = 'Strong' THEN 2 ELSE 1 END < 2 ORDER BY dp.VolGB * CASE WHEN dp.DataSense = 'High' THEN 3 WHEN dp.DataSense = 'Medium' THEN 2 ELSE 1 END DESC LIMIT 5;
