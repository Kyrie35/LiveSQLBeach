WITH signals AS ( SELECT CAST(( market_sentiment_indicators -> '$.oscillators' ->> '$.rsi14spot' ) AS REAL) AS rsi14spot, CAST(( market_sentiment_indicators -> '$.oscillators' ->> '$.macdtrail' ) AS REAL) AS macdtrail, CAST(( market_sentiment_indicators -> '$.oscillators' ->> '$.bbandspan' ) AS REAL) AS bbandspan, market_sentiment_indicators -> '$.momentum' ->> '$.techmeter' AS techmeter FROM analyticsindicators ) SELECT rsi14spot, macdtrail, bbandspan, techmeter, ( ( ABS(rsi14spot - 50) + ABS(macdtrail) + ( bbandspan * 0.01 ) ) / 3 ) * CASE WHEN techmeter = 'Buy' THEN 1 WHEN techmeter = 'Sell' THEN -1 ELSE 0 END AS technical_signal_strength FROM signals;
