SELECT o.opsregistry, o.emerglevel, ROUND( ( CAST(hr.staffingprofile -> '$.personnel' ->> '$.total' AS REAL) / ( f.personnelcostsusd / 10000.0 ) ) * ( CAST(hr.staffingprofile -> '$.readiness' ->> '$.availability_percent' AS REAL) / 100.0 ), 2 ) AS per FROM operations AS o JOIN humanresources AS hr ON o.opsregistry = hr.hropsref JOIN financials AS f ON o.opsregistry = f.finopsref WHERE o.emerglevel IN ('Red', 'Black');
