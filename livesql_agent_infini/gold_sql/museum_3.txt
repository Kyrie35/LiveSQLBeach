WITH status_mapping AS ( SELECT ArtRegistry, ArtName, ArtDynasty, CASE ConserveStatus WHEN 'Excellent' THEN 1 WHEN 'Good' THEN 3 WHEN 'Fair' THEN 5 WHEN 'Poor' THEN 7 WHEN 'Critical' THEN 10 END AS status_score FROM ArtifactsCore ) SELECT a.ArtRegistry, a.ArtName, a.ArtDynasty, ( ( r.HistSignRating + r.ResearchValRating + r.CultScore ) * ( 10 - s.status_score ) ) / 30.0 AS CPI FROM status_mapping AS s JOIN ArtifactRatings AS r ON s.ArtRegistry = r.ArtRef JOIN ArtifactsCore AS a ON s.ArtRegistry = a.ArtRegistry ORDER BY CPI DESC NULLS FIRST;
