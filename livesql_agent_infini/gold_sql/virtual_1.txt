WITH gift_metrics AS ( SELECT f.userregistry, f.nicklabel, i.giftvalusd, i.gifttot, ( i.giftvalusd * i.gifttot / 100.0 ) AS gift_impact_quotient, ms.spendusd, ms.membkind FROM fans AS f JOIN interactions AS i ON i.interactfanpivot = f.userregistry JOIN membershipandspending AS ms ON ms.memberfanpivot = f.userregistry ) SELECT userregistry, nicklabel, ROUND(gift_impact_quotient, 2) AS giq, membkind FROM gift_metrics WHERE gift_impact_quotient > 50 AND spendusd < 100 AND membkind IN ('Free', 'Basic') ORDER BY gift_impact_quotient DESC NULLS FIRST;
