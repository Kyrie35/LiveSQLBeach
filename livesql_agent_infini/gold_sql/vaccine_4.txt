WITH ShipmentRisk AS ( SELECT s.ShipmentRegistry, CASE WHEN s.IntegrityMark = 'Compromised' OR s.SealFlag = 'Broken' OR s.TamperSign = 'Confirmed' THEN 'Compromised' ELSE 'Safe' END AS risk_status, COALESCE(rm.Incidents, 0) AS incident_count FROM Shipments s LEFT JOIN RegulatoryAndMaintenance rm ON s.ShipmentRegistry = rm.ShipGov ) SELECT ShipmentRegistry, risk_status, incident_count, GROUP_CONCAT(rm.RegMaintRegistry) AS maintenance_ids FROM ShipmentRisk sr LEFT JOIN RegulatoryAndMaintenance rm ON sr.ShipmentRegistry = rm.ShipGov WHERE risk_status = 'Compromised' GROUP BY ShipmentRegistry, risk_status, incident_count ORDER BY incident_count DESC LIMIT 50;
