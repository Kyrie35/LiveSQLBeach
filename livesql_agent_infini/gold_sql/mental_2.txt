WITH FacilityResources AS ( SELECT fackey, CASE json_extract(support_and_resources, '$.community_resources') WHEN 'Comprehensive' THEN 3 WHEN 'Adequate' THEN 2 WHEN 'Limited' THEN 1 ELSE 0 END AS resource_score FROM facilities WHERE support_and_resources IS NOT NULL ), FacilityAdherence AS ( SELECT e.facid, COUNT(*) AS total_outcomes, SUM(CASE WHEN tro.txadh IN ('High', 'Medium') THEN 1 ELSE 0 END) AS adherent_outcomes FROM treatmentoutcomes tro JOIN treatmentbasics tb ON tro.txref = tb.txkey JOIN encounters e ON tb.encref = e.enckey WHERE e.facid IS NOT NULL GROUP BY e.facid ), FacilityMetrics AS ( SELECT fr.fackey, fr.resource_score, CASE WHEN fa.total_outcomes > 0 THEN CAST(fa.adherent_outcomes AS REAL) / fa.total_outcomes ELSE 0 END AS tar FROM FacilityResources fr LEFT JOIN FacilityAdherence fa ON fr.fackey = fa.facid ), OverallFRAI AS ( SELECT AVG(resource_score) AS frai FROM FacilityResources ), CorrelationData AS ( SELECT (SELECT frai FROM OverallFRAI) AS overall_frai, resource_score, tar FROM FacilityMetrics WHERE tar > 0 ) SELECT overall_frai, ( SELECT SUM((resource_score - (SELECT AVG(resource_score) FROM CorrelationData)) * (tar - (SELECT AVG(tar) FROM CorrelationData))) / (SQRT(SUM((resource_score - (SELECT AVG(resource_score) FROM CorrelationData)) * (resource_score - (SELECT AVG(resource_score) FROM CorrelationData)))) * SQRT(SUM((tar - (SELECT AVG(tar) FROM CorrelationData)) * (tar - (SELECT AVG(tar) FROM CorrelationData))))) FROM CorrelationData ) AS correlation_frai_tar FROM (SELECT DISTINCT overall_frai FROM CorrelationData);
