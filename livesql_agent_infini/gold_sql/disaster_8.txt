SELECT f.financeregistry, o.opsregistry, d.distregistry, ROUND( ( f.donorcommitmentsusd / f.budgetallotusd ) * ( 1 - f.fundsutilpct / 100 ) - ( f.resourcegapsusd / f.budgetallotusd ), 2 ) AS financial_sustainability_ratio, CASE WHEN f.fundsutilpct > 80 AND f.fundingstate = 'Critical' AND ( ( f.donorcommitmentsusd / f.budgetallotusd ) * ( 1 - f.fundsutilpct / 100 ) - ( f.resourcegapsusd / f.budgetallotusd ) ) < 0.2 THEN 'Financial Crisis Risk' ELSE 'Financially Stable' END AS financial_status FROM financials AS f JOIN operations AS o ON f.finopsref = o.opsregistry JOIN disasterevents AS d ON f.findistref = d.distregistry ORDER BY financial_sustainability_ratio NULLS LAST;
