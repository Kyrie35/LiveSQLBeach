WITH influence_calc AS ( SELECT a.accindex, m.netinflscore, cb.postfreq, COUNT(*) OVER (PARTITION BY m.clustsize) AS cluster_member_count FROM account AS a JOIN profile AS p ON p.profaccref = a.accindex JOIN sessionbehavior AS sb ON sb.sessprofref = p.profkey JOIN contentbehavior AS cb ON cb.cntsessref = sb.sessref JOIN moderationaction AS m ON m.macntref = cb.cntref WHERE m.clustrole = 'Amplifier' AND m.coordscore > 0.8 ), amplifiers AS ( SELECT accindex, netinflscore, postfreq, cluster_member_count, DENSE_RANK() OVER (ORDER BY netinflscore DESC NULLS FIRST) AS influence_rank FROM influence_calc WHERE netinflscore > 80 AND postfreq > 30 ) SELECT accindex, netinflscore, postfreq, cluster_member_count, influence_rank FROM amplifiers WHERE influence_rank <= 10 LIMIT 100;
