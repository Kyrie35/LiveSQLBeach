WITH TraderBaseMetrics AS ( SELECT t.tradereg, t.freqscope, t.acctbal, t.voldaily / NULLIF(t.acctbal, 0) AS dtr, CAST(( t.trading_performance -> '$.risklevel' ->> '$.levratio' ) AS REAL) AS tle, ( t.trading_performance -> '$.risklevel' ->> '$.risklevel' ) AS risk_level_text FROM trader AS t ), HighRiskProfileTraders AS ( SELECT tradereg, acctbal, freqscope FROM TraderBaseMetrics WHERE tle > 5.0 AND risk_level_text = 'Aggressive' OR dtr > 0.5 ), HighFreqHighRiskTraders AS ( SELECT tradereg, acctbal FROM HighRiskProfileTraders WHERE freqscope = 'High' ), EnforcementImpact AS ( SELECT ea.enforcereg, ea.traderestr, ea.penamt / NULLIF(hfhrt.acctbal, 0) AS efir, CASE WHEN ea.penimp IN ('Fine', 'Ban') OR ea.acttake = 'Suspension' OR ea.busrestr = 'Full' THEN TRUE ELSE FALSE END AS is_significant_action FROM enforcementactions AS ea JOIN compliancecase AS cc ON ea.compref2 = cc.compreg JOIN transactionrecord AS tr ON cc.transref = tr.transreg JOIN HighFreqHighRiskTraders AS hfhrt ON tr.trdref = hfhrt.tradereg ), FinanciallyImpactfulCases AS ( SELECT enforcereg, traderestr FROM EnforcementImpact WHERE is_significant_action = TRUE AND efir > 0.1 ), CostlyHighFreqCases AS ( SELECT fic.enforcereg, fic.traderestr FROM FinanciallyImpactfulCases AS fic ) SELECT enforcereg, traderestr FROM CostlyHighFreqCases;
