SELECT v.vendregistry, v.vendrate, m.mktdenom AS market, ( CAST(v.vendsucccount AS REAL) / v.vendtxcount * 100 ) - ( CAST(v.venddisputecount AS REAL) / v.vendtxcount * 50 ) + ( v.vendrate * 5 ) AS vti FROM vendors AS v JOIN markets AS m ON v.mktref = m.mktregistry WHERE vendchecklvl IN ('Advanced', 'Premium') AND ( CAST(v.venddisputecount AS REAL) / v.vendtxcount ) < 0.05 AND v.vendspan > 90 AND ( ( CAST(v.vendsucccount AS REAL) / v.vendtxcount * 100 ) - ( CAST(v.venddisputecount AS REAL) / v.vendtxcount * 50 ) + ( v.vendrate * 5 ) ) > 80 ORDER BY vti DESC NULLS FIRST;
