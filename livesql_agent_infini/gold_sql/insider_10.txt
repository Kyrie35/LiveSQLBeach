WITH TransactionBaseMetrics AS ( SELECT tr.transreg, tr.risk_indicators, ( t.voldaily / NULLIF(t.acctbal, 0) ) AS dtr, CAST(( t.trading_performance -> '$.risklevel' ->> '$.levratio' ) AS REAL) AS tle, ( tr.modfreq / NULLIF(1 - tr.cancelpct, 0) ) AS omi FROM transactionrecord AS tr JOIN trader AS t ON tr.trdref = t.tradereg WHERE tr.cancelpct < 1 ), CalculatedScores AS ( SELECT tbm.transreg, tbm.risk_indicators ->> '$.layerind' AS layer_indicator, ( tbm.dtr * tbm.tle * tbm.omi ) AS ati, ( ( 0.3 * CAST(( tbm.risk_indicators ->> '$.spoofprob' ) AS REAL) / 100.0 ) + ( 0.2 * CAST(( tbm.risk_indicators ->> '$.frontscore' ) AS REAL) / 100.0 ) + ( 0.1 * CAST(( tbm.risk_indicators ->> '$.qstuffindex' ) AS REAL) ) + ( 0.2 * CASE tbm.risk_indicators ->> '$.washsus' WHEN 'High' THEN 1.0 WHEN 'Medium' THEN 0.5 ELSE 0.1 END ) + ( 0.2 * CASE tbm.risk_indicators ->> '$.layerind' WHEN 'Confirmed' THEN 1.0 WHEN 'Suspected' THEN 0.5 ELSE 0.0 END ) ) AS sai FROM TransactionBaseMetrics AS tbm ), ASS_Calculation AS ( SELECT cs.transreg, cs.layer_indicator, ( cs.sai * cs.ati ) AS ass FROM CalculatedScores AS cs WHERE cs.layer_indicator IN ('Confirmed', 'Suspected') ) SELECT layer_indicator, ROUND(AVG(ass), 3) AS average_ass FROM ASS_Calculation GROUP BY layer_indicator;
