WITH bbi_calc AS ( SELECT a.accindex AS macntref, a.acctident, ( 0.4 * COALESCE( CAST(( sd.detection_score_profile -> '$.behavior_scores' ->> '$.botlikscore' ) AS REAL) / 100, 0 ) + 0.3 * COALESCE( CAST(( sd.detection_score_profile -> '$.behavior_scores' ->> '$.autobehavscore' ) AS REAL), 0 ) + 0.3 * ( 1 - COALESCE(ma.convnatval, 0) ) ) AS bbi_score FROM account AS a LEFT JOIN moderationaction AS m ON a.accindex = m.macntref LEFT JOIN securitydetection AS sd ON m.masedetref = sd.secdetkey LEFT JOIN messaginganalysis AS ma ON m.macntref = ma.msgcntref WHERE a.acctstatus = 'Dormant' ) SELECT bc.acctident, bc.bbi_score, CASE WHEN bc.bbi_score > 0.9 THEN 'High Risk' WHEN bc.bbi_score > 0.3 THEN 'Moderate Risk' ELSE 'Low Risk' END AS risk_category FROM bbi_calc AS bc WHERE bc.bbi_score > 0.2 ORDER BY bc.bbi_score DESC NULLS FIRST LIMIT 100;
