WITH vendor_disputes AS ( SELECT m.mktregistry, COALESCE(SUM(v.venddisputecount), 0) AS total_disputes, COALESCE(SUM(v.vendtxcount), 1) AS total_transactions FROM markets AS m LEFT JOIN vendors AS v ON m.mktregistry = v.mktref GROUP BY m.mktregistry ) SELECT m.mktregistry, m.mktdenom, m.mktclass, ( m.mktspan / 365.0 ) * ( m.esccomprate / 100.0 ) * ( 1 - COALESCE(CAST(vd.total_disputes AS REAL) / NULLIF(vd.total_transactions, 0), 0) ) * 100 AS msi, m.market_status_reputation ->> '$.status' AS status FROM markets AS m JOIN vendor_disputes AS vd ON m.mktregistry = vd.mktregistry ORDER BY msi DESC NULLS FIRST;
