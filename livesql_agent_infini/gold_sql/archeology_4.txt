SELECT sr.logregistry AS registration_id, sr.arcref AS project_id, sr.logaccumm AS registration_accuracy_mm, sr.errvalmm AS error_value_mm, ROUND( ( spc.scanresolmm / ( sr.logaccumm * SQRT(1.0 + sr.errvalmm / sr.logaccumm) ) ), 2 ) AS rar, sr.logmethod, CASE WHEN ( spc.scanresolmm / ( sr.logaccumm * SQRT(1.0 + sr.errvalmm / sr.logaccumm) ) ) > 1.5 AND sr.refmark LIKE '%Target%' THEN 'High Confidence' WHEN ( spc.scanresolmm / ( sr.logaccumm * SQRT(1.0 + sr.errvalmm / sr.logaccumm) ) ) BETWEEN 1.0 AND 1.5 THEN 'Medium Confidence' ELSE 'Low Confidence' END AS confidence_level FROM scanregistration AS sr JOIN scanpointcloud AS spc ON sr.arcref = spc.arcref;
