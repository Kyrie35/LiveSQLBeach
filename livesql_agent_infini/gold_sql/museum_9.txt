WITH erf_calc AS ( SELECT sd.artref, ( CASE envsensitivity WHEN 'Low' THEN 1 WHEN 'Medium' THEN 5 WHEN 'High' THEN 10 ELSE 5 END + CASE lightsensitivity WHEN 'Low' THEN 1 WHEN 'Medium' THEN 5 WHEN 'High' THEN 10 ELSE 5 END + CASE tempsensitivity WHEN 'Low' THEN 1 WHEN 'Medium' THEN 5 WHEN 'High' THEN 10 ELSE 5 END + CASE humiditysensitivity WHEN 'Low' THEN 1 WHEN 'Medium' THEN 5 WHEN 'High' THEN 10 ELSE 5 END ) / 4.0 AS erf FROM sensitivitydata AS sd ), ler_calc AS ( SELECT ca.artrefexamined AS artref, ( lr.lightlux * CASE sd.lightsensitivity WHEN 'Low' THEN 1 WHEN 'Medium' THEN 5 WHEN 'High' THEN 10 ELSE 5 END * lr.visibleexplxh ) / 1000.0 AS ler FROM conditionassessments AS ca JOIN lightandradiationreadings AS lr ON ca.lightreadrefobserved = lr.lightradregistry JOIN sensitivitydata AS sd ON ca.artrefexamined = sd.artref ), mdr_calc AS ( SELECT ac.artregistry AS artref, ( ac.artageyears * POWER(( ec.relhumidity - 50 ), 2) * ec.tempc * erf.erf ) / 100000.0 AS mdr FROM artifactscore AS ac JOIN sensitivitydata AS sd ON ac.artregistry = sd.artref JOIN erf_calc AS erf ON erf.artref = sd.artref JOIN environmentalreadingscore AS ec ON ec.showcaseref = ( SELECT ur.showcaserefused FROM usagerecords AS ur WHERE ur.artrefused = ac.artregistry LIMIT 1 ) ), tetl_calc AS ( SELECT e.artref, ROUND(CAST(( e.erf + COALESCE(l.ler, 0) + COALESCE(m.mdr, 0) * 2 ) AS REAL), 2) AS tetl FROM erf_calc AS e LEFT JOIN ler_calc AS l ON e.artref = l.artref LEFT JOIN mdr_calc AS m ON e.artref = m.artref ) SELECT * FROM tetl_calc ORDER BY tetl DESC NULLS FIRST LIMIT 10;
