WITH subscription_value AS ( SELECT userkey, typelabel, substatus, subdays, ageval, gendlbl, occulbl, subdays * CASE substatus WHEN 'Premium' THEN 2.0 WHEN 'Enterprise' THEN 3.0 WHEN 'Basic' THEN 1.0 ELSE 0.5 END AS usv_score FROM users WHERE substatus IS NOT NULL AND subdays > 0 ), demographic_score AS ( SELECT sv.userkey, sv.typelabel, sv.substatus, sv.ageval, sv.gendlbl, sv.occulbl, sv.usv_score, sv.ageval * CASE sv.gendlbl WHEN 'M' THEN 1.0 WHEN 'F' THEN 1.0 ELSE 0.8 END * CASE sv.occulbl WHEN 'Professional' THEN 1.5 WHEN 'Student' THEN 0.7 WHEN 'Retired' THEN 1.2 ELSE 1.0 END AS uds_score FROM subscription_value sv ), user_interests AS ( SELECT userkey, json_extract(user_preferences, '$.interests') AS interest_list, CAST(json_extract(user_preferences, '$.satisfaction_score') AS REAL) AS satisfaction_score FROM users WHERE json_extract(user_preferences, '$.interests') IS NOT NULL ), interest_split AS ( SELECT userkey, interest_list, satisfaction_score, '' AS interest, interest_list || ',' AS remaining, 0 AS level FROM user_interests WHERE interest_list IS NOT NULL UNION ALL SELECT userkey, interest_list, satisfaction_score, TRIM(substr(remaining, 1, instr(remaining, ',') - 1)) AS interest, substr(remaining, instr(remaining, ',') + 1) AS remaining, level + 1 FROM interest_split WHERE remaining <> '' AND level < 20 ), interests_unnested AS ( SELECT userkey, interest_list, satisfaction_score, interest FROM interest_split WHERE interest <> '' ) SELECT ds.substatus, ds.gendlbl, ds.occulbl, CASE WHEN ds.ageval < 25 THEN 'Under 25' WHEN ds.ageval BETWEEN 25 AND 34 THEN '25-34' WHEN ds.ageval BETWEEN 35 AND 44 THEN '35-44' WHEN ds.ageval BETWEEN 45 AND 54 THEN '45-54' ELSE '55+' END AS age_group, COUNT(DISTINCT ds.userkey) AS user_count, ROUND(AVG(ds.usv_score), 2) AS avg_subscription_value, ROUND(AVG(ds.uds_score), 2) AS avg_demographic_score, ROUND(AVG(ui.satisfaction_score), 2) AS avg_satisfaction, GROUP_CONCAT(DISTINCT iu.interest) AS common_interests FROM demographic_score ds LEFT JOIN user_interests ui ON ds.userkey = ui.userkey LEFT JOIN interests_unnested iu ON ds.userkey = iu.userkey GROUP BY ds.substatus, ds.gendlbl, ds.occulbl, CASE WHEN ds.ageval < 25 THEN 'Under 25' WHEN ds.ageval BETWEEN 25 AND 34 THEN '25-34' WHEN ds.ageval BETWEEN 35 AND 44 THEN '35-44' WHEN ds.ageval BETWEEN 45 AND 54 THEN '45-54' ELSE '55+' END HAVING COUNT(DISTINCT ds.userkey) > 10 ORDER BY avg_subscription_value DESC, avg_demographic_score DESC LIMIT 15;
