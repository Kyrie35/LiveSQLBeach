SELECT f.userregistry, f.nicklabel, CAST(( sc.community_engagement -> '$.content_creation' ->> '$.contqualrate' ) AS REAL) AS content_quality, CAST(( sc.community_engagement -> '$.content_creation' ->> '$.ugcval' ) AS INTEGER) AS content_volume, CAST(( sc.community_engagement -> '$.network' ->> '$.follcount' ) AS INTEGER) AS follower_count, CASE WHEN CAST(( sc.community_engagement -> '$.content_creation' ->> '$.contqualrate' ) AS REAL) > 8.5 AND CAST(( sc.community_engagement -> '$.network' ->> '$.follcount' ) AS INTEGER) > 1000 AND CAST(( sc.community_engagement -> '$.content_creation' ->> '$.ugcval' ) AS INTEGER) > 20 THEN 'High-Value Content Creator' WHEN CAST(( sc.community_engagement -> '$.content_creation' ->> '$.contqualrate' ) AS REAL) > 7.0 AND CAST(( sc.community_engagement -> '$.content_creation' ->> '$.ugcval' ) AS INTEGER) > 20 THEN 'Content Creator' ELSE 'Regular Fan' END AS creator_classification FROM fans AS f JOIN membershipandspending AS ms ON ms.memberfanpivot = f.userregistry JOIN engagement AS e ON e.engagememberpivot = ms.memberreg JOIN socialcommunity AS sc ON sc.socialengagepivot = e.engagereg WHERE NOT CAST(( sc.community_engagement -> '$.content_creation' ->> '$.contqualrate' ) AS REAL) IS NULL ORDER BY content_quality DESC NULLS FIRST, follower_count DESC NULLS FIRST;
