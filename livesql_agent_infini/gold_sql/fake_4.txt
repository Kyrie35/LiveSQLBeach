WITH content_manipulation AS ( SELECT cb.cntref, cb.cntsessref, ( 0.4 * ( 1 - cb.cntuniqscore ) + 0.3 * cb.mediareratio + 0.3 * ( 1 - m.txtuniq ) ) AS cms FROM contentbehavior AS cb JOIN messaginganalysis AS m ON m.msgcntref = cb.cntref ), content_farms AS ( SELECT cm.cntref, cm.cntsessref, cm.cms, cb.postfreq, CASE WHEN cb.postfreq > 50 THEN 'High' WHEN cb.postfreq > 20 THEN 'Medium' ELSE 'Low' END AS freq_category, DENSE_RANK() OVER (ORDER BY cm.cms DESC NULLS FIRST) AS manipulation_rank FROM content_manipulation AS cm JOIN contentbehavior AS cb ON cb.cntref = cm.cntref WHERE cm.cms > 0.7 AND cb.postfreq > 50 ) SELECT cntref, cms, postfreq, freq_category, manipulation_rank FROM content_farms WHERE manipulation_rank <= 10 ORDER BY manipulation_rank NULLS LAST LIMIT 100;
