WITH article_readability AS ( SELECT artkey, catlabel, readsec * LOG(wordlen) / CASE difflevel WHEN 'Basic' THEN 1 WHEN 'Intermediate' THEN 1.5 WHEN 'Advanced' THEN 2 ELSE 1.2 END AS ars, CAST(REPLACE(REPLACE(JSON_EXTRACT(engagement_metrics, '$.engagement_rate'), '{', ''), '}', '') AS REAL) AS engagement_rate FROM articles WHERE wordlen > 0 AND readsec > 0 ), CorrelationData AS ( SELECT catlabel, AVG(ars) AS avg_ars, AVG(engagement_rate) AS avg_engagement_rate, SUM(ars * engagement_rate) AS sum_ars_engagement, COUNT(*) AS n, SUM(ars * ars) AS sum_ars_squared, SUM(engagement_rate * engagement_rate) AS sum_engagement_squared FROM article_readability GROUP BY catlabel ) SELECT cd.catlabel, cd.avg_ars AS avg_readability, cd.avg_engagement_rate AS avg_engagement, ( cd.n * cd.sum_ars_engagement - SUM(ar.ars) * SUM(ar.engagement_rate) ) / ( SQRT( (cd.n * cd.sum_ars_squared - SUM(ar.ars) * SUM(ar.ars)) * (cd.n * cd.sum_engagement_squared - SUM(ar.engagement_rate) * SUM(ar.engagement_rate)) ) ) AS readability_engagement_corr, SUM(CASE WHEN ar.ars < 50 THEN 1 ELSE 0 END) AS low_readability, SUM(CASE WHEN ar.ars BETWEEN 50 AND 100 THEN 1 ELSE 0 END) AS medium_readability, SUM(CASE WHEN ar.ars > 100 THEN 1 ELSE 0 END) AS high_readability FROM article_readability ar JOIN CorrelationData cd ON ar.catlabel = cd.catlabel GROUP BY cd.catlabel;
