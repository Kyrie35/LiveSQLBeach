WITH session_analysis AS ( SELECT s.seshkey, s.seshstart, s.bncrate, s.ctrval, s.seshviews, s.engscore, s.seshdur, s.bncrate * ( 1 - s.ctrval / 100.0 ) AS sbra, AVG(i.seqval) OVER (PARTITION BY s.seshkey) AS avg_seqval FROM sessions AS s JOIN interactions AS i ON s.seshkey = i.seshlink2 ) SELECT seshkey, seshstart, sbra, avg_seqval, avg_seqval / NULLIF(sbra, 0) AS rtse FROM session_analysis WHERE NOT sbra IS NULL;
