WITH aaf_calc AS ( SELECT a.accindex, ( CAST(sb.sesscount AS REAL) / a.acctagespan ) AS aaf FROM account AS a JOIN profile AS p ON p.profaccref = a.accindex JOIN sessionbehavior AS sb ON sb.sessprofref = p.profkey ), ngv_calc AS ( SELECT a.accindex, SQRT( POWER( CAST(( n.network_engagement_metrics -> '$.connection_metrics' ->> '$.followgrowrate' ) AS REAL), 2 ) + POWER( CAST(( n.network_engagement_metrics -> '$.connection_metrics' ->> '$.followinggrowrate' ) AS REAL), 2 ) ) AS ngv FROM account AS a JOIN profile AS p ON p.profaccref = a.accindex JOIN sessionbehavior AS sb ON sb.sessprofref = p.profkey JOIN networkmetrics AS n ON n.netsessref = sb.sessref ), bbi_calc AS ( SELECT a.accindex, ( 0.4 * ( CAST(( s.detection_score_profile -> '$.behavior_scores' ->> '$.botlikscore' ) AS REAL) / 100 ) + 0.3 * CAST(( s.detection_score_profile -> '$.behavior_scores' ->> '$.autobehavscore' ) AS REAL) + 0.3 * ( 1 - m.convnatval ) ) AS bbi FROM account AS a JOIN profile AS p ON p.profaccref = a.accindex JOIN sessionbehavior AS sb ON sb.sessprofref = p.profkey JOIN contentbehavior AS cb ON cb.cntsessref = sb.sessref JOIN messaginganalysis AS m ON m.msgcntref = cb.cntref JOIN moderationaction AS ma ON ma.macntref = cb.cntref JOIN securitydetection AS s ON ma.masedetref = s.secdetkey WHERE CAST(( s.detection_score_profile -> '$.behavior_scores' ->> '$.botlikscore' ) AS REAL) > 70 ), bas_calc AS ( SELECT a.accindex, ROUND(( 0.4 * bbi.bbi + 0.4 * aa.aaf + 0.2 * ng.ngv ), 2) AS bas, FIRST_VALUE(s.detection_score_profile -> '$.behavior_scores' ->> '$.botlikscore') OVER (PARTITION BY a.accindex ORDER BY s.detecttime DESC NULLS FIRST) AS latest_botlikscore FROM account AS a JOIN profile AS p ON p.profaccref = a.accindex JOIN sessionbehavior AS sb ON sb.sessprofref = p.profkey JOIN contentbehavior AS cb ON cb.cntsessref = sb.sessref JOIN messaginganalysis AS m ON m.msgcntref = cb.cntref JOIN moderationaction AS ma ON ma.macntref = cb.cntref JOIN securitydetection AS s ON ma.masedetref = s.secdetkey JOIN aaf_calc AS aa ON aa.accindex = a.accindex JOIN ngv_calc AS ng ON ng.accindex = a.accindex JOIN bbi_calc AS bbi ON bbi.accindex = a.accindex WHERE CAST(( s.detection_score_profile -> '$.behavior_scores' ->> '$.botlikscore' ) AS REAL) > 70 ) SELECT accindex, bas, latest_botlikscore FROM bas_calc WHERE bas > 0.8 ORDER BY bas DESC NULLS FIRST LIMIT 10;
