WITH SafetyScores AS ( SELECT ps.effectivenessrobot, SUM( COALESCE(CAST(( safety_metrics ->> '$.overloads' ) AS INTEGER), 0) + COALESCE(CAST(( safety_metrics ->> '$.collisions' ) AS INTEGER), 0) + COALESCE(CAST(( safety_metrics ->> '$.emergency_stops' ) AS INTEGER), 0) + COALESCE(CAST(( safety_metrics ->> '$.speed_violations' ) AS INTEGER), 0) ) AS sis FROM performance_and_safety AS ps GROUP BY ps.effectivenessrobot ), RecentFaults AS ( SELECT mf.upkeeprobot, mf.faultpredscore, ROW_NUMBER() OVER (PARTITION BY mf.upkeeprobot ORDER BY mf.upkeepduedays ASC NULLS LAST) AS rn FROM maintenance_and_fault AS mf WHERE NOT mf.upkeepduedays IS NULL ) SELECT s.effectivenessrobot, s.sis, rf.faultpredscore AS recent_fault_score FROM SafetyScores AS s INNER JOIN RecentFaults AS rf ON s.effectivenessrobot = rf.upkeeprobot WHERE s.sis > 20 AND rf.rn = 1 AND rf.faultpredscore > 0.5 ORDER BY s.sis DESC NULLS FIRST LIMIT 100;
