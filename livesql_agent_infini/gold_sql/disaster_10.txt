WITH coordination_factor_mapping AS ( SELECT b.beneregistry, b.benedistref, b.beneopsref, b.distequityidx, c.coordeffectlvl, CASE WHEN c.coordeffectlvl = 'High' THEN 1.2 WHEN c.coordeffectlvl = 'Medium' THEN 1.0 WHEN c.coordeffectlvl = 'Low' THEN 0.8 ELSE 0 END AS coordeffect_factor FROM beneficiariesandassessments AS b JOIN coordinationandevaluation AS c ON b.benedistref = c.coorddistref AND b.beneopsref = c.coordopsref ), rde_calculation AS ( SELECT cf.beneregistry, cf.benedistref, cf.beneopsref, cf.distequityidx, cf.coordeffectlvl, cf.coordeffect_factor, t.distributionpoints, t.deliverysuccessrate, ROUND( cf.distequityidx * ( 1 + ( t.distributionpoints / 20.0 ) ) * ( 1 - ( ( 100 - t.deliverysuccessrate ) / 100.0 ) ) * cf.coordeffect_factor, 2 ) AS rde FROM coordination_factor_mapping AS cf JOIN transportation AS t ON cf.benedistref = t.transportdistref ) SELECT r.beneregistry, r.rde, CASE WHEN r.rde < 0.4 AND r.distequityidx < 0.5 AND r.distributionpoints < 5 THEN 'Resource Distribution Inequity' ELSE 'Normal Distribution' END AS distribution_status, d.impactmetrics -> '$.population' ->> '$.affected' AS affected_population FROM rde_calculation AS r JOIN disasterevents AS d ON r.benedistref = d.distregistry ORDER BY r.rde NULLS LAST;
