WITH system_alerts AS ( SELECT sp.rectrace, sp.resptime, sp.loadscore, sp.perfscore, ( sp.perfscore - sp.loadscore ) AS performance_delta, RANK() OVER (ORDER BY sp.resptime DESC NULLS FIRST) AS rt_rank FROM systemperformance AS sp WHERE sp.resptime > 100 ) SELECT rectrace, resptime, loadscore, perfscore, performance_delta, rt_rank, CASE WHEN resptime > 200 THEN 'Critical' WHEN resptime > 150 THEN 'Warning' ELSE 'Normal' END AS performance_status FROM system_alerts;
