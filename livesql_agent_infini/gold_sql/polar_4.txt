WITH renewable_contribution AS ( SELECT tswg.thermaleqref, l.stationname, l.locationtype, tswg.renewablemetrics ->> '$.solar' AS solar_json, tswg.renewablemetrics ->> '$.wind' AS wind_json, tswg.fuelcelloutputw, CAST(( tswg.renewablemetrics -> '$.solar' ->> '$.output_w' ) AS REAL) + CAST(( tswg.renewablemetrics -> '$.wind' ->> '$.output_w' ) AS REAL) AS renewable_output, ( CAST(( tswg.renewablemetrics -> '$.solar' ->> '$.output_w' ) AS REAL) + CAST(( tswg.renewablemetrics -> '$.wind' ->> '$.output_w' ) AS REAL) ) / NULLIF( ( tswg.fuelcelloutputw + CAST(( tswg.renewablemetrics -> '$.solar' ->> '$.output_w' ) AS REAL) + CAST(( tswg.renewablemetrics -> '$.wind' ->> '$.output_w' ) AS REAL) ), 0 ) * 100 AS rec FROM thermalsolarwindandgrid AS tswg JOIN equipment AS e ON tswg.thermaleqref = e.equipmentcode JOIN location AS l ON e.equipmentcode = l.loceqref WHERE NOT ( tswg.renewablemetrics -> '$.solar' ->> '$.output_w' ) IS NULL OR NOT ( tswg.renewablemetrics -> '$.wind' ->> '$.output_w' ) IS NULL ) SELECT locationtype, stationname, COUNT(*) AS equipment_count, ROUND(CAST(rec AS REAL), 2) AS rec, ROUND(CAST(SUM(renewable_output) AS REAL), 2) AS total_renewable_output_w, CASE WHEN rec > 70 THEN 'Energy-Sustainable' WHEN rec > 50 THEN 'Moderately Sustainable' ELSE 'Low Sustainability' END AS sustainability_classification FROM renewable_contribution GROUP BY locationtype, stationname, rec ORDER BY rec DESC NULLS FIRST;
