WITH fault_metrics AS ( SELECT upkeeprobot, SUM(faultpredscore * ( 1.0 / ( 1 + upkeepduedays ) )) / SUM(1.0 / ( 1 + upkeepduedays )) AS wfps, MIN(rulhours) AS mrul FROM maintenance_and_fault GROUP BY upkeeprobot ), operation_stress AS ( SELECT operbotdetref, MAX(totopshrval) AS toh, COUNT(*) AS operation_count FROM operation GROUP BY operbotdetref ) SELECT rd.botdetreg, fm.wfps, fm.mrul, os.toh, os.operation_count, CASE WHEN fm.wfps > 0.6 AND fm.mrul < 200 THEN 'CRITICAL' WHEN fm.wfps > 0.4 OR fm.mrul < 500 THEN 'WARNING' ELSE 'NORMAL' END AS maintenance_priority FROM robot_details AS rd JOIN fault_metrics AS fm ON rd.botdetreg = fm.upkeeprobot JOIN operation_stress AS os ON rd.botdetreg = os.operbotdetref WHERE os.toh > 1000 ORDER BY fm.wfps DESC NULLS FIRST, fm.mrul ASC NULLS LAST LIMIT 100;
