WITH evasion_scores AS ( SELECT t.techkey, a.accindex, ROUND((0.4 * t.vpnratio + 0.3 * (t.proxycount / 10.0) + 0.3 * (t.ipcountrynum / 20.0)), 2) AS tei, t.ipcountrynum FROM technicalinfo t JOIN networkmetrics n ON n.netkey = t.technetref JOIN sessionbehavior sb ON sb.sessref = n.netsessref JOIN profile p ON p.profkey = sb.sessprofref JOIN account a ON a.accindex = p.profaccref WHERE t.vpnratio > 0 GROUP BY t.techkey, a.accindex, t.vpnratio, t.proxycount, t.ipcountrynum ), vpn_abusers AS ( SELECT e.accindex, e.tei, e.ipcountrynum, ( SELECT COUNT(*) FROM (SELECT DISTINCT tei FROM evasion_scores ORDER BY tei) AS distinct_teis WHERE tei <= e.tei ) * 4 / (SELECT COUNT(DISTINCT tei) FROM evasion_scores) + 1 AS tei_quartile FROM evasion_scores e WHERE e.tei > 0.8 AND e.ipcountrynum >= 3 ) SELECT accindex, tei, ipcountrynum, tei_quartile FROM vpn_abusers WHERE tei_quartile = 4 ORDER BY tei DESC LIMIT 10;
