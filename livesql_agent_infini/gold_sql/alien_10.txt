WITH stability_analysis AS ( SELECT s.SignalRegistry, s.SignalClass, s.FreqDriftHzs, s.FreqMhz, s.SigDurSec, s.DoppShiftHz, s.ModType, s.ModIndex, sd.SigIntegrity, sd.FreqStab, sd.PhaseStab, sc.EntropyVal, ((1 - ABS(s.FreqDriftHzs)/(s.FreqMhz * 1000)) * (s.SigDurSec/(1 + s.DoppShiftHz/1000))) AS SSM FROM Signals s INNER JOIN SignalDynamics sd ON s.SignalRegistry = sd.SignalRef INNER JOIN SignalClassification sc ON s.SignalRegistry = sc.SignalRef ), stability_attrs AS ( SELECT SignalRegistry, 'FreqStab' AS attribute, FreqStab AS value FROM stability_analysis UNION ALL SELECT SignalRegistry, 'PhaseStab' AS attribute, PhaseStab AS value FROM stability_analysis UNION ALL SELECT SignalRegistry, 'SigIntegrity' AS attribute, SigIntegrity AS value FROM stability_analysis ) SELECT sa.SignalRegistry, sa.SignalClass, ROUND(sa.SSM, 3) AS SSM, GROUP_CONCAT(CASE WHEN sa.attribute = 'FreqStab' THEN 'Frequency: ' || sa.value WHEN sa.attribute = 'PhaseStab' THEN 'Phase: ' || sa.value WHEN sa.attribute = 'SigIntegrity' THEN 'Integrity: ' || sa.value END, ', ') AS stability_factors, CASE WHEN sa.SSM > 0.8 AND sa.EntropyVal BETWEEN 0.4 AND 0.8 AND sa.ModIndex > 0.5 THEN 'Coherent Information Pattern Detected' WHEN sa.SSM > 0.6 AND sa.EntropyVal BETWEEN 0.3 AND 0.9 THEN 'Potential Information Pattern' ELSE 'No Clear Pattern' END AS pattern_assessment FROM ( SELECT sa.SignalRegistry, sa.SignalClass, sa.SSM, sa.EntropyVal, sa.ModType, sa.ModIndex, stability_attrs.attribute, stability_attrs.value FROM stability_analysis sa JOIN stability_attrs ON sa.SignalRegistry = stability_attrs.SignalRegistry ) AS sa GROUP BY sa.SignalRegistry, sa.SignalClass, sa.SSM, sa.EntropyVal, sa.ModType, sa.ModIndex;
