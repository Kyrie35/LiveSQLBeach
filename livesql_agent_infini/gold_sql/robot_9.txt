WITH PositionError AS ( SELECT ad.actdetref, rd.ctrltypeval, AVG(ad.poserrmmval) AS ape FROM actuation_data ad JOIN robot_details rd ON ad.actdetref = rd.botdetreg GROUP BY ad.actdetref, rd.ctrltypeval ), FaultScore AS ( SELECT mf.upkeeprobot, MAX(mf.faultpredscore) AS rfps FROM maintenance_and_fault mf GROUP BY mf.upkeeprobot ), RobotAge AS ( SELECT rd.botdetreg, (JULIANDAY(rr.rects) - JULIANDAY(rd.instdateval)) / 365.25 AS ray FROM robot_details rd JOIN robot_record rr ON rd.botdetreg = rr.recreg ), Ranked AS ( SELECT pe.actdetref, pe.ctrltypeval, pe.ape, CAST(RANK() OVER (PARTITION BY pe.ctrltypeval ORDER BY pe.ape DESC) AS REAL) / COUNT(*) OVER (PARTITION BY pe.ctrltypeval) AS ape_rank FROM PositionError pe ) SELECT r.ctrltypeval AS controller_type, AVG(r.ape) AS avg_ape, MAX(r.ape_rank) AS max_ape_rank, AVG(COALESCE(fs.rfps, 0)) AS avg_rfps, COUNT(DISTINCT r.actdetref) AS robot_count FROM Ranked r LEFT JOIN FaultScore fs ON r.actdetref = fs.upkeeprobot INNER JOIN RobotAge ra ON r.actdetref = ra.botdetreg WHERE r.ape >= 0.1 AND COALESCE(fs.rfps, 0) > 0.5 AND ra.ray > 1 GROUP BY r.ctrltypeval ORDER BY max_ape_rank DESC LIMIT 50;
