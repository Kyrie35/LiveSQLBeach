WITH influence_metrics AS ( SELECT a.accindex, ROUND( ( 0.4 * CAST(( nm.network_engagement_metrics -> '$.connection_metrics' ->> '$.connqualscore' ) AS REAL) + 0.3 * m.netinflscore + 0.3 * CAST(( nm.network_engagement_metrics -> '$.interaction_metrics' ->> '$.interactdiv' ) AS REAL) / 100 ), 2 ) AS nic, RANK() OVER (ORDER BY ( 0.4 * CAST(( nm.network_engagement_metrics -> '$.connection_metrics' ->> '$.connqualscore' ) AS REAL) + 0.3 * m.netinflscore + 0.3 * CAST(( nm.network_engagement_metrics -> '$.interaction_metrics' ->> '$.interactdiv' ) AS REAL) / 100 ) DESC NULLS FIRST) AS influence_rank FROM account AS a JOIN profile AS p ON p.profaccref = a.accindex JOIN sessionbehavior AS sb ON sb.sessprofref = p.profkey JOIN networkmetrics AS nm ON nm.netsessref = sb.sessref JOIN contentbehavior AS cb ON cb.cntsessref = sb.sessref JOIN moderationaction AS m ON m.macntref = cb.cntref ) SELECT accindex, nic, influence_rank FROM influence_metrics WHERE nic > 0 ORDER BY influence_rank NULLS LAST LIMIT 10;
