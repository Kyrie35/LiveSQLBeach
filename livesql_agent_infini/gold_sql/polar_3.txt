WITH ors_calc AS ( SELECT om.opmainteqref, CASE WHEN om.operationalstatus = 'Active' THEN 10 * (1 - om.operationhours / om.maintenancecyclehours) WHEN om.operationalstatus = 'Standby' THEN 5 * (1 - om.operationhours / om.maintenancecyclehours) ELSE 0 END AS ors FROM operationmaintenance om ), tie_calc AS ( SELECT thermaleqref, CASE WHEN insulationstatus = 'Good' THEN 0.9 - (heatlossratekwh / 10) WHEN insulationstatus = 'Fair' THEN 0.6 - (heatlossratekwh / 10) WHEN insulationstatus = 'Poor' THEN 0.3 - (heatlossratekwh / 10) ELSE 0 END AS tie FROM thermalsolarwindandgrid ), lssr_calc AS ( SELECT oc.opmainteqref, 0.7 * oc.ors + 0.3 * COALESCE(tc.tie, 0) AS lssr FROM ors_calc oc LEFT JOIN tie_calc tc ON oc.opmainteqref = tc.thermaleqref ) SELECT e.equipmentcode, e.equipmenttype, ls.lifesupportstatus, ROUND(lc.lssr, 2) AS lssr, CASE WHEN lc.lssr >= 0.8 THEN 'High Reliability' WHEN lc.lssr >= 0.6 THEN 'Moderate Reliability' ELSE 'Low Reliability' END AS reliability_classification, json_object( 'oxygen_status', ls.oxygensupplystatus, 'medical_status', ls.medicalequipmentstatus, 'safety_system_status', ls.safetysystemstatus ) AS support_systems_status FROM equipment e JOIN lssr_calc lc ON e.equipmentcode = lc.opmainteqref JOIN lightingandsafety ls ON e.equipmentcode = ls.lighteqref WHERE e.equipmenttype = 'Safety' ORDER BY lssr DESC;
