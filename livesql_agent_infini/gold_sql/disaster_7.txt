WITH engagement_mapping AS ( SELECT b.beneregistry, b.benedistref, b.commengagelvl, b.vulnerabilityreview, b.benefeedbackscore, b.distequityidx, c.stakeholdersatisf, CASE WHEN b.commengagelvl = 'Low' THEN 1 WHEN b.commengagelvl = 'Medium' THEN 2 WHEN b.commengagelvl = 'High' THEN 3 ELSE 0 END AS commengage_numeric, ( b.benefeedbackscore * 10 ) + ( CASE WHEN b.commengagelvl = 'Low' THEN 1 WHEN b.commengagelvl = 'Medium' THEN 2 WHEN b.commengagelvl = 'High' THEN 3 ELSE 0 END * 20 ) + ( b.distequityidx * 50 ) AS bsi FROM beneficiariesandassessments AS b JOIN coordinationandevaluation AS c ON b.benedistref = c.coorddistref AND b.beneopsref = c.coordopsref ), cee_calculation AS ( SELECT e.*, ROUND( ( e.bsi / 100 ) * e.commengage_numeric * ( ( e.stakeholdersatisf + 1 ) / 5 ), 2 ) AS cee, CASE WHEN b.localcapacitygrowth = 'Active' AND b.commengagelvl = 'High' AND e.bsi > 70 THEN 'Community Resilience Builder' WHEN ( e.bsi / 100 ) * e.commengage_numeric * ( ( e.stakeholdersatisf + 1 ) / 5 ) > 2.5 AND e.vulnerabilityreview = 'Complete' AND NOT ( b.localcapacitygrowth = 'Active' AND b.commengagelvl = 'High' AND e.bsi > 70 ) THEN 'Community Resilience Opportunity' ELSE 'Standard Operation' END AS community_status FROM engagement_mapping AS e JOIN beneficiariesandassessments AS b ON e.beneregistry = b.beneregistry ) SELECT c.beneregistry, c.benedistref, c.cee, c.community_status, RANK() OVER (PARTITION BY c.community_status ORDER BY c.cee DESC NULLS FIRST) AS rank_within_status FROM cee_calculation AS c ORDER BY c.community_status NULLS LAST, c.cee DESC NULLS FIRST;
