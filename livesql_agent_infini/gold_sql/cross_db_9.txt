WITH IntegrityIssues AS ( SELECT profiletrace, intcheck, csumverify, (CASE WHEN intcheck = 'Failed' THEN 1 ELSE 0 END) + (CASE WHEN csumverify = 'Failed' THEN 1 ELSE 0 END) AS ifc FROM dataprofile WHERE intcheck = 'Failed' OR csumverify = 'Failed' ) SELECT profiletrace, ifc AS integrity_failure_count, TRIM((CASE WHEN intcheck = 'Failed' THEN 'Integrity Check, ' ELSE '' END || CASE WHEN csumverify = 'Failed' THEN 'Checksum Verification, ' ELSE '' END), ', ') AS failures FROM IntegrityIssues ORDER BY ifc DESC LIMIT 10;
