SELECT si.zoneregistry AS site_code, si.zonelabel AS site_name, ROUND( CAST(( 0.3 * COALESCE(adc.adc, 0) ) + ( 0.3 * COALESCE(mfs.mfs, 0) ) + ( 0.2 * COALESCE(rar.rar, 0) ) + ( 0.2 * COALESCE(sce.sce, 0) ) - ( 2 * SQRT(COALESCE(err.error_ratio, 0)) ) AS REAL), 2 ) AS dpq FROM sites AS si LEFT JOIN ( SELECT s.zoneref, AVG( ( sqs.sqs * 0.4 ) + ( sm.mfs * 0.4 ) + ( spc.coverpct * ( 1.0 + spc.lappct / 100.0 * ( 1.0 - spc.coverpct / 100.0 ) ) * 0.2 ) - ( 5 * SQRT(spc.noisedb / 10.0) ) ) AS adc FROM scans AS s JOIN scanpointcloud AS spc ON s.arcref = spc.arcref LEFT JOIN ( SELECT sc.arcref, AVG( POWER(10.0 / ( LOG(10, sc.scanresolmm * 1000.0) / LOG(10, sc.pointdense) ) * 5.0, 1.5) * ( sc.coverpct * ( 1.0 + sc.lappct / 100.0 * ( 1.0 - sc.coverpct / 100.0 ) ) / 100.0 ) * POWER(1.0 - sc.noisedb / 30.0, 2) ) AS sqs FROM scanpointcloud AS sc GROUP BY sc.arcref ) AS sqs ON s.arcref = sqs.arcref LEFT JOIN ( SELECT sm.zoneref, AVG( ( sm.facetfaces / ( sm.facetverts * sm.facetresmm * sm.facetresmm ) * 1000.0 ) * ( sm.texpix / ( SQRT(sm.facetfaces) * sm.facetresmm ) * 0.01 / 10.0 ) * ( 1.0 + EXP(-sm.geomdeltamm) ) ) AS mfs FROM scanmesh AS sm WHERE sm.facetverts > 0 AND sm.facetfaces > 0 AND sm.facetresmm > 0 GROUP BY sm.zoneref ) AS sm ON s.zoneref = sm.zoneref GROUP BY s.zoneref ) AS adc ON si.zoneregistry = adc.zoneref LEFT JOIN ( SELECT sm.zoneref, AVG( ( sm.facetfaces / ( sm.facetverts * sm.facetresmm * sm.facetresmm ) * 1000.0 ) * ( sm.texpix / ( SQRT(sm.facetfaces) * sm.facetresmm ) * 0.01 / 10.0 ) * ( 1.0 + EXP(-sm.geomdeltamm) ) ) AS mfs FROM scanmesh AS sm WHERE sm.facetverts > 0 AND sm.facetfaces > 0 AND sm.facetresmm > 0 GROUP BY sm.zoneref ) AS mfs ON si.zoneregistry = mfs.zoneref LEFT JOIN ( SELECT s.zoneref, AVG(spc.scanresolmm / ( sr.logaccumm * SQRT(1.0 + sr.errvalmm / sr.logaccumm) )) AS rar FROM scanregistration AS sr JOIN scanpointcloud AS spc ON sr.arcref = spc.arcref JOIN scans AS s ON sr.arcref = s.arcref WHERE sr.logaccumm > 0 GROUP BY s.zoneref ) AS rar ON si.zoneregistry = rar.zoneref LEFT JOIN ( SELECT s.zoneref, AVG( spc.coverpct * ( 1.0 + spc.lappct / 100.0 * ( 1.0 - spc.coverpct / 100.0 ) ) ) AS sce FROM scans AS s JOIN scanpointcloud AS spc ON s.arcref = spc.arcref GROUP BY s.zoneref ) AS sce ON si.zoneregistry = sce.zoneref LEFT JOIN ( SELECT s.zoneref, AVG(sr.errvalmm / spc.scanresolmm) AS error_ratio FROM scanregistration AS sr JOIN scanpointcloud AS spc ON sr.arcref = spc.arcref JOIN scans AS s ON sr.arcref = s.arcref WHERE spc.scanresolmm > 0 GROUP BY s.zoneref ) AS err ON si.zoneregistry = err.zoneref ORDER BY dpq DESC NULLS FIRST;
