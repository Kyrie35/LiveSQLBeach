WITH soiling_data AS ( SELECT e.arearegistry, e.envmoment, e.soillosspct, e.dustdengm2, AVG(e.soillosspct / e.dustdengm2) OVER ( PARTITION BY e.arearegistry ) as avg_sif FROM environment e ), cleaning_schedule AS ( SELECT sd.arearegistry, sd.envmoment, sd.soillosspct, sd.dustdengm2, sd.avg_sif, LAG( sd.envmoment, 1, DATE(sd.envmoment, '-30 days') ) OVER ( PARTITION BY sd.arearegistry ORDER BY sd.envmoment ) as prev_moment FROM soiling_data sd ) SELECT cs.arearegistry, cs.envmoment, cs.soillosspct, cs.dustdengm2, cs.avg_sif FROM cleaning_schedule cs WHERE cs.soillosspct > 5 OR cs.dustdengm2 > 0.15 OR ( cs.avg_sif * cs.dustdengm2 ) > 0.15 OR JULIANDAY(cs.envmoment) - JULIANDAY(cs.prev_moment) > 30;
