WITH MaterialDeterioration AS ( SELECT art.ArtRegistry, art.ArtName, sens.SensitivityRegistry, sens.EnvSensitivity, sens.LightSensitivity, sens.TempSensitivity, sens.HumiditySensitivity, sens.VibraSensitivity, sens.PollutantSensitivity, sens.PestSensitivity, ( CASE WHEN sens.EnvSensitivity = 'High' THEN 1 ELSE 0 END + CASE WHEN sens.LightSensitivity = 'High' THEN 1 ELSE 0 END + CASE WHEN sens.TempSensitivity = 'High' THEN 1 ELSE 0 END + CASE WHEN sens.HumiditySensitivity = 'High' THEN 1 ELSE 0 END + CASE WHEN sens.VibraSensitivity = 'High' THEN 1 ELSE 0 END + CASE WHEN sens.PollutantSensitivity = 'High' THEN 1 ELSE 0 END + CASE WHEN sens.PestSensitivity = 'High' THEN 1 ELSE 0 END ) AS high_sensitivity_count FROM artifactscore art JOIN sensitivitydata sens ON art.ArtRegistry = sens.ArtRef ), LatestReadings AS ( SELECT env.ShowcaseRef, env.TempC, env.RelHumidity, env.ReadTimestamp FROM environmentalreadingscore env WHERE env.EnvReadRegistry IN (SELECT MAX(EnvReadRegistry) FROM environmentalreadingscore GROUP BY ShowcaseRef) ), ArtifactMDR AS ( SELECT md.ArtRegistry, md.ArtName, md.high_sensitivity_count, ur.ShowcaseRefUsed, lr.TempC, lr.RelHumidity, (art.ArtAgeYears * CASE WHEN md.EnvSensitivity = 'Low' THEN 1 WHEN md.EnvSensitivity = 'Medium' THEN 5 WHEN md.EnvSensitivity = 'High' THEN 10 END * POWER((lr.RelHumidity - 50), 2) * lr.TempC) / 100000.0 AS mdr FROM MaterialDeterioration md JOIN artifactscore art ON md.ArtRegistry = art.ArtRegistry JOIN usagerecords ur ON md.ArtRegistry = ur.ArtRefUsed LEFT JOIN LatestReadings lr ON ur.ShowcaseRefUsed = lr.ShowcaseRef ) SELECT mdr.ArtRegistry, mdr.ArtName, mdr.TempC AS current_temp, mdr.RelHumidity AS current_humidity, mdr.high_sensitivity_count, CASE WHEN mdr.mdr > 5 AND mdr.high_sensitivity_count >= 2 THEN 'Yes' ELSE 'No' END AS accelerated_deterioration FROM ArtifactMDR mdr;
