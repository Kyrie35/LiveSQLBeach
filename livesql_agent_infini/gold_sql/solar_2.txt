SELECT p.growalias AS plant_name, p.gencapmw AS capacity_mw, SUM(m.maintcostusd) AS total_maintenance_costs, SUM(m.revlossusd) AS revenue_losses, AVG(m.maintcostusd / p.gencapmw) AS maintenance_cost_efficiency_score, AVG(m.revlossusd / p.gencapmw) AS revenue_loss_rate, AVG((m.revlossusd - m.maintcostusd) / p.gencapmw) AS mroi_value, CASE WHEN AVG((m.revlossusd - m.maintcostusd) / p.gencapmw) > 100 THEN 'URGENT' WHEN AVG((m.revlossusd - m.maintcostusd) / p.gencapmw) > 50 THEN 'HIGH' WHEN AVG((m.revlossusd - m.maintcostusd) / p.gencapmw) > 0 THEN 'MEDIUM' ELSE 'LOW' END AS final_priority_level FROM plant AS p LEFT JOIN maintenance AS m ON p.growregistry = m.powerref GROUP BY p.growalias, p.gencapmw ORDER BY CASE WHEN final_priority_level = 'URGENT' THEN 1 WHEN final_priority_level = 'HIGH' THEN 2 WHEN final_priority_level = 'MEDIUM' THEN 3 ELSE 4 END, mroi_value DESC;
