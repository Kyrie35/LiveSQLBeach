WITH plant_maintenance_costs AS ( SELECT p.growalias, p.gencapmw, SUM(m.maintcostusd + m.cleancostusd + m.replacecostusd) AS total_maintenance_cost, AVG(m.revlossusd) AS avg_rev_loss FROM plant AS p JOIN maintenance AS m ON m.powerref = p.growregistry GROUP BY p.growalias, p.gencapmw ), maint_efficiency AS ( SELECT growalias, gencapmw, total_maintenance_cost / gencapmw AS mce, avg_rev_loss / gencapmw AS rlr FROM plant_maintenance_costs ) SELECT growalias, mce, rlr, rlr / mce AS mroi FROM maint_efficiency;
