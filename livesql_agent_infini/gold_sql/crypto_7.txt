WITH money_flows AS ( SELECT CAST(( md.quote_depth_snapshot -> '$.metadata' ->> '$.timetrack' ) AS TIMESTAMP) AS flow_time, ( md.quote_depth_snapshot -> '$.metadata' ->> '$.mktcombo' ) AS market_pair, CAST(( CAST(( a.market_sentiment_indicators ->> '$.flow' ) AS JSONB) ->> '$.smartforce' ) AS REAL) AS smart_force, CAST(( CAST(( a.market_sentiment_indicators ->> '$.flow' ) AS JSONB) ->> '$.retailflow' ) AS REAL) AS retail_flow, CAST(( CAST(( a.market_sentiment_indicators ->> '$.flow' ) AS JSONB) ->> '$.instflow' ) AS REAL) AS inst_flow, ( CAST(( a.market_sentiment_indicators ->> '$.big_players' ) AS JSONB) ->> '$.whalemotion' ) AS whale_activity, CAST(( md.quote_depth_snapshot -> '$.quotes' ->> '$.midquote' ) AS REAL) AS mid_price FROM analyticsindicators AS a JOIN marketdata AS md ON a.mdataref = md.marketdatanode ), price_changes AS ( SELECT mf.*, LEAD(mid_price, 1) OVER (PARTITION BY market_pair ORDER BY flow_time NULLS LAST) AS next_price_1h, LEAD(mid_price, 4) OVER (PARTITION BY market_pair ORDER BY flow_time NULLS LAST) AS next_price_4h, LEAD(mid_price, 24) OVER (PARTITION BY market_pair ORDER BY flow_time NULLS LAST) AS next_price_24h, CASE WHEN smart_force > retail_flow * 1.2 AND smart_force > inst_flow * 1.2 THEN 'Smart Money Dominant' WHEN retail_flow > smart_force * 1.2 AND retail_flow > inst_flow * 1.2 THEN 'Retail Dominant' WHEN inst_flow > smart_force * 1.2 AND inst_flow > retail_flow * 1.2 THEN 'Institutional Dominant' ELSE 'Mixed' END AS flow_dominance FROM money_flows AS mf ) SELECT flow_dominance, whale_activity, AVG(next_price_1h - mid_price) AS avg_1h_change, AVG(next_price_4h - mid_price) AS avg_4h_change, AVG(next_price_24h - mid_price) AS avg_24h_change, CAST(SUM( CASE WHEN ( smart_force > retail_flow AND smart_force > inst_flow AND next_price_4h > mid_price ) OR ( smart_force < retail_flow AND smart_force < inst_flow AND next_price_4h < mid_price ) THEN 1 ELSE 0 END ) AS REAL) / COUNT(*) AS smart_money_accuracy FROM price_changes WHERE NOT next_price_24h IS NULL GROUP BY flow_dominance, whale_activity, market_pair ORDER BY smart_money_accuracy DESC NULLS FIRST;
