WITH TherapyEngagement AS ( SELECT e.facid, tb.txkey, CASE ( tb.therapy_details ->> '$.engagement' ) WHEN 'High' THEN 3 WHEN 'Medium' THEN 2 WHEN 'Low' THEN 1 WHEN 'Non-compliant' THEN 0 ELSE 0 END AS engagement_score FROM treatmentbasics AS tb JOIN encounters AS e ON tb.encref = e.enckey WHERE NOT tb.therapy_details IS NULL AND NOT e.facid IS NULL ), FacilityTES AS ( SELECT facid, ROUND(AVG(engagement_score), 2) AS avg_tes FROM TherapyEngagement GROUP BY facid ), FunctionalImprovement AS ( SELECT e.facid, tro.txoutkey, CASE tro.funcimpv WHEN 'Significant' THEN 3 WHEN 'Moderate' THEN 2 WHEN 'Minimal' THEN 1 ELSE 0 END AS funcimpv_score, CASE WHEN tro.txadh IN ('High', 'Medium') THEN 1 ELSE 0 END AS is_adherent FROM treatmentoutcomes AS tro JOIN treatmentbasics AS tb ON tro.txref = tb.txkey JOIN encounters AS e ON tb.encref = e.enckey WHERE NOT e.facid IS NULL ), FacilityRTI AS ( SELECT facid, AVG(funcimpv_score) AS avg_funcimpv_score, CASE WHEN COUNT(*) > 0 THEN CAST(SUM(is_adherent) AS REAL) / COUNT(*) ELSE 0 END AS tar, ROUND( ( AVG(funcimpv_score) * CASE WHEN COUNT(*) > 0 THEN CAST(SUM(is_adherent) AS REAL) / COUNT(*) ELSE 0 END ), 2 ) AS rti FROM FunctionalImprovement GROUP BY facid ) SELECT ft.facid, ft.avg_tes, fr.rti FROM FacilityTES AS ft JOIN FacilityRTI AS fr ON ft.facid = fr.facid WHERE ft.avg_tes > 2.0 AND fr.rti < 0.8 ORDER BY ft.facid NULLS LAST LIMIT 100;
