SELECT a.audregistry, d.makername, d.modnum, a.thdpct, a.spksensedb, a.audlatms, ROUND( ( 1 - a.thdpct / 2.0 ) * ( a.spksensedb / 100.0 ) * ( 1 - a.audlatms / 100.0 ) * 10, 2 ) AS aqi FROM audioandmedia AS a JOIN deviceidentity AS d ON a.auddevref = d.devregistry WHERE ( 1 - a.thdpct / 2.0 ) * ( a.spksensedb / 100.0 ) * ( 1 - a.audlatms / 100.0 ) * 10 > 8.0 ORDER BY aqi DESC NULLS FIRST;
